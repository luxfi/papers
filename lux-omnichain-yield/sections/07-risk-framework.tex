\section{Risk Framework}
\label{sec:risk}
% ============================================================

\subsection{Three-Tier Classification}

\begin{definition}[Risk Tiers]
Each yield strategy is classified into one of three risk tiers:
\begin{itemize}[nosep]
  \item \textbf{Tier 1 (Low)}: Blue-chip protocols with \$1B+ TVL, multiple audits, 2+ years of production operation, no exploits. Examples: Lido, Aave V3, Compound V3, Curve, Spark, Marinade.
  \item \textbf{Tier 2 (Medium)}: Established protocols with \$100M+ TVL, audited, 1+ year of production operation. Examples: EigenLayer, Symbiotic, Morpho, Pendle, Babylon, Lombard, Jito, Kamino, Tonstakers, CoreDAO.
  \item \textbf{Tier 3 (High)}: Newer protocols with elevated yield and correspondingly elevated risk. Examples: Ethena, SolvBTC, STON.fi, LPX Perps.
\end{itemize}
\end{definition}

\subsection{Allocation Constraints}

The vault enforces per-tier allocation limits:

\begin{equation}\label{eq:allocation}
\begin{aligned}
  w_1 + w_2 + w_3 &= 1 \\
  w_1 &\geq 0.50 \\
  w_2 &\leq 0.40 \\
  w_3 &\leq 0.10
\end{aligned}
\end{equation}

where $w_i$ is the fraction of total backing allocated to Tier $i$ strategies. These constraints are enforced on-chain by the \texttt{YieldBridgeVault} contract.

\begin{theorem}[Diversification Bound]\label{thm:diversification}
No single strategy may receive more than $\alpha_{\max} = 25\%$ of total backing. Combined with the tier constraints, this ensures:
\begin{equation}
  \max_j \frac{A_j}{B} \leq \alpha_{\max} = 0.25
\end{equation}
where $A_j$ is the assets deployed to strategy $j$ and $B = \sum_j A_j$ is total backing.
\end{theorem}

\subsection{Maximum Loss Analysis}

\begin{definition}[Strategy Loss]
Let $\ell_j$ denote the worst-case loss fraction for strategy $j$ (e.g., smart contract exploit draining all funds: $\ell_j = 1.0$). The maximum portfolio loss under the allocation constraints is:
\begin{equation}\label{eq:maxloss}
  L_{\max} = \max_{\substack{w \in \mathcal{W} \\ j^* \in \{1,\ldots,m\}}} w_{j^*} \cdot \ell_{j^*}
\end{equation}
where $\mathcal{W}$ is the feasible allocation set satisfying~(\ref{eq:allocation}).
\end{definition}

Under the single-strategy cap of 25\%:

\begin{corollary}[Single-Strategy Failure]
If exactly one strategy suffers a complete loss ($\ell_{j^*} = 1.0$), the maximum portfolio loss is 25\%. Combined with the 1.5\% collateralization buffer, the bridge pauses before share price drops below $0.985 \times p(t_{k-1})$, protecting holders from losses exceeding 1.5\% between attestation intervals.
\end{corollary}

\begin{corollary}[Tier 3 Failure]
If all Tier 3 strategies simultaneously fail ($\ell_j = 1.0$ for all $j$ in Tier 3), the maximum loss is 10\% of total backing. The bridge pauses immediately at the next attestation.
\end{corollary}

\subsection{Strategy Health Monitoring}

Each strategy's \texttt{isActive()} function is checked at every attestation cycle. A strategy is deactivated when:
\begin{enumerate}[nosep]
  \item \texttt{totalAssets()} decreases by more than 5\% between attestations (abnormal loss).
  \item The underlying protocol's TVL drops below \$50M (liquidity risk).
  \item A security advisory is issued (governance override via SAB or MPC emergency action).
  \item \texttt{currentAPY()} drops below 0\% for two consecutive attestation periods.
\end{enumerate}

Upon deactivation, the vault withdraws all assets from the strategy and reallocates according to the tier-weighted algorithm.
