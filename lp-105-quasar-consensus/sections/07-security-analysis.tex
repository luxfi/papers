\section{Security Analysis}

\label{sec:security}
%% ============================================================================

We state four theorems. Full proofs are provided in the Lean~4 formalization
(\texttt{Quasar.lean}, \texttt{BFT.lean}, \texttt{BLS.lean},
\texttt{Corona.lean}, \texttt{MLDSA.lean}) and the Tamarin symbolic model
(\texttt{QuasarConsensus.spthy}). Here we give proof sketches.

\subsection{Safety}

\begin{theorem}[Safety]
\label{thm:safety}
If fewer than $f+1$ validators are Byzantine (i.e., $f < n/3$), then no two
conflicting blocks $B_h$ and $B_h'$ at the same height $h$ can both obtain
valid Quasar certificates.
\end{theorem}

\begin{proof}[Proof sketch]
A Quasar certificate requires $\geq 2f+1$ signatures from each of the three
layers. Consider the BLS layer. The set of BLS signers for $B_h$ is
$Q^{\mathrm{BLS}}_h$ with $|Q^{\mathrm{BLS}}_h| \geq 2f+1$. Similarly for
$B_h'$: $|Q^{\mathrm{BLS}}_{h'}| \geq 2f+1$. The total number of validators
is $n = 3f + \delta$ where $\delta \geq 1$.

The intersection satisfies:
\[
|Q^{\mathrm{BLS}}_h \cap Q^{\mathrm{BLS}}_{h'}| \geq
2(2f+1) - n = 2(2f+1) - (3f+\delta) = f + 2 - \delta \geq 1
\]

We use the form $2(2f+1) - n$ rather than $\frac{2 \cdot 2f+1 - n}{1}$ to
avoid integer truncation. Since $n \leq 3f+1$ for the tightest case
($\delta = 1$), the intersection contains at least $f+1$ validators. At least
one of these is correct (since at most $f$ are Byzantine). A correct validator
never signs two conflicting blocks at the same height. Contradiction.

The same argument applies independently to the Corona quorum (threshold
$t = 2f+1$ contributors needed) and the ML-DSA quorum ($2f+1$ individual
signatures). Each layer independently prevents conflicting certificates.
\end{proof}

\subsection{Liveness}

\begin{theorem}[Liveness]
\label{thm:liveness}
After GST, if the leader for height $h$ is correct, then a valid quantum
certificate for some block at height $h$ is produced within bounded time
$3\Delta + T_{\mathrm{sign}}$ where $T_{\mathrm{sign}}$ is the maximum
signing latency across the three layers.
\end{theorem}

\begin{proof}[Proof sketch]
After GST, a correct leader's proposal reaches all correct validators within
$\Delta$. Correct validators execute Wave (FPC rounds complete in bounded time
since $\theta_{\max} < 1$ and $\beta$ is finite). Once $\geq 2f+1$ correct
validators lock their preference, they produce and exchange signatures. Since
$n - f \geq 2f+1$, at least $2f+1$ correct validators exist and will
contribute signatures to all three layers. Message delivery after GST takes at
most $\Delta$. The Corona two-round protocol completes in $2\Delta$.
BLS and ML-DSA signing is non-interactive (each validator signs independently).
Total: proposal delivery ($\Delta$) + Wave convergence (bounded) + Corona
rounds ($2\Delta$) + certificate assembly. The dominant term is $3\Delta +
T_{\mathrm{sign}}$.
\end{proof}

\subsection{Quasar Unforgeability}

\begin{theorem}[Quasar Unforgeability]
\label{thm:unforgeability}
Forging a Quasar certificate requires simultaneously breaking: (1) the
co-CDH assumption on BLS12-381, (2) Module-SIS for Corona, and (3)
Module-LWE for ML-DSA-65. The probability of forgery is:
\[
\Pr[\mathrm{Forge}] \leq \epsilon_{\mathrm{BLS}} +
\epsilon_{\mathrm{RT}} + \epsilon_{\mathrm{ML}} -
\epsilon_{\mathrm{BLS}} \cdot \epsilon_{\mathrm{RT}} -
\epsilon_{\mathrm{BLS}} \cdot \epsilon_{\mathrm{ML}} -
\epsilon_{\mathrm{RT}} \cdot \epsilon_{\mathrm{ML}} +
\epsilon_{\mathrm{BLS}} \cdot \epsilon_{\mathrm{RT}} \cdot
\epsilon_{\mathrm{ML}}
\]
where $\epsilon_X$ is the probability of breaking layer $X$. Since the
certificate requires all three layers, an adversary must forge signatures
in every layer:
\[
\Pr[\mathrm{Forge}_{\mathrm{quantum}}] \leq
\epsilon_{\mathrm{BLS}} \cdot \epsilon_{\mathrm{RT}} \cdot
\epsilon_{\mathrm{ML}}
\]
\end{theorem}

\begin{proof}[Proof sketch]
A Quasar certificate $\mathcal{C}_h$ is valid only if all three components
verify. An adversary without $f+1$ corrupted validators must forge at least
one signature per layer to produce a conflicting certificate. The three
forgery problems are independent (different hardness assumptions, different
algebraic structures). By a standard hybrid argument, an adversary that
forges the Quasar certificate can be used to construct a forger for at least
one of the three underlying schemes. The joint probability of independent
forgeries is their product.
\end{proof}

\subsection{Single-Compromise Insufficiency}

\begin{theorem}[Single-Compromise Insufficiency]
\label{thm:single}
An adversary that breaks exactly one of the three cryptographic layers
cannot produce a valid conflicting Quasar certificate, provided $f < n/3$
honest validators remain.
\end{theorem}

\begin{proof}[Proof sketch]
Case analysis on which layer is compromised.

\textbf{Case 1: BLS broken.} The adversary can forge BLS signatures. However,
a valid Quasar certificate also requires a Corona threshold signature from
$\geq t$ share holders and $\geq 2f+1$ valid ML-DSA-65 signatures. The
adversary controls at most $f$ Corona shares and $f$ ML-DSA keys.
Since $f < t$ and $f < 2f+1$, the adversary cannot produce the Corona
or ML-DSA components.

\textbf{Case 2: Corona broken.} The adversary can forge Corona threshold
signatures. It still needs $\geq 2f+1$ valid BLS signatures and $\geq 2f+1$
valid ML-DSA signatures. With at most $f$ corrupted keys in each layer,
this is impossible.

\textbf{Case 3: ML-DSA broken.} Symmetric to Case~2: forging ML-DSA
signatures does not help with BLS or Corona.

In each case, the two intact layers enforce the quorum intersection argument
of Theorem~\ref{thm:safety}.
\end{proof}


\subsection{QuasarCert Soundness}

\begin{theorem}[QuasarCert Soundness]
\label{thm:qcert-sound}
Let $\mathcal{C} = (\sigma_{\mathrm{RT}}, \sigma_{\mathrm{BLS}},
\pi_{\mathrm{ZK}})$ be a QuasarCert that verifies for block $B$ under
validator set $\mathcal{V}$ with $|\mathcal{V}| = n$ and threshold
$t = \lfloor 2n/3 \rfloor + 1$.  Then under the co-CDH assumption, the
Module-LWE assumption, the MLWE/MSIS assumptions, and the KEA/AGM
assumptions:

\medskip\noindent
At least $t$ validators in $\mathcal{V}$ signed block $B$.  Equivalently,
if $f < n/3$ validators are Byzantine, then no QuasarCert for a block
$B' \neq B$ not agreed upon by honest validators can pass verification,
except with probability $\mathsf{negl}(\lambda)$.

\medskip\noindent\textit{Note: This theorem assumes static corruption.
Extension to adaptive corruption is proved in the companion proof sketch
(Appendices C--D of~\cite{qcert-proof}).  The Groth16 circuit
specification and R1CS constraint analysis appear in Appendix~B
of~\cite{qcert-proof}.}
\end{theorem}

\begin{proof}
We show that forging any single component of QuasarCert requires
breaking its underlying assumption.  Since all three must verify
(conjunction, Def.~\ref{def:verify-qcert}), forging the full
certificate requires breaking all three simultaneously.

\medskip
\textbf{Component 1: $\sigma_{\mathrm{RT}}$ (Corona).}
By the EUF-CMA theorem for Corona~\cite{corona}, the threshold
signature scheme is existentially unforgeable under Module-LWE.
In the threshold setting with $t = \lfloor 2n/3\rfloor+1$, an adversary
controlling $f < n/3 < t$ parties cannot produce a valid Corona
signature on a message not signed by the honest quorum.  The advantage
is bounded by:
\[
\mathsf{Adv}^{\text{EUF-CMA}}_{\text{RT}} \leq Q_H \cdot
\mathsf{Adv}^{\text{MLWE}} + Q_S / 2^{128}
\]
If $\sigma_{\mathrm{RT}}$ verifies for $H(B)$, then at least $t$
validators contributed shares.

\medskip
\textbf{Component 2: $\sigma_{\mathrm{BLS}}$ (BLS aggregate).}
By Theorem~1 of Boneh--Lynn--Shacham~\cite{boneh2001short}, BLS
signatures are EUF-CMA secure under co-CDH in the random oracle model.
An aggregate BLS signature over a set $S \subseteq \mathcal{V}$ verifies
iff every $v_i \in S$ produced a valid individual signature.  The
rogue-key attack is prevented by proof-of-possession.  For the aggregate
to verify with the expected quorum key, $|S| \geq t$.

\medskip
\textbf{Component 3: $\pi_{\mathrm{ZK}}$ (Groth16 of ML-DSA).}
The Groth16 proof attests: ``there exist $n$ valid ML-DSA signatures on
$H(B)$ under the validator public keys.''  By knowledge-soundness of
Groth16 in the AGM~\cite{groth2016}, if $\pi_{\mathrm{ZK}}$ verifies,
an extractor can recover the $n$ ML-DSA signatures.  Each extracted
signature is valid under $\mathit{pk}_i^{\mathrm{ML}}$, which by
EUF-CMA of ML-DSA (FIPS~204, MLWE/MSIS~\cite{nist-pqc}) implies
validator $v_i$ produced it.

\medskip
\textbf{Composition.}
A valid QuasarCert for $B$ implies:
\begin{itemize}[nosep]
  \item $\geq t$ validators signed via Corona (or MLWE is broken),
  \item $\geq t$ validators signed via BLS (or co-CDH is broken),
  \item all $n$ validators signed via ML-DSA (or MLWE/MSIS or KEA is
    broken).
\end{itemize}
The intersection of the first two signing sets has size
$\geq 2t - n \geq n/3 + 2 > f + 1$, so at least one honest validator
is in both.  Combined with the ML-DSA attestation (all honest
validators), the certificate is unforgeable unless all four assumptions
fail.  By a union bound:
\[
\mathsf{Adv}^{\mathsf{forge}} \leq
\mathsf{Adv}^{\text{co-CDH}} + \mathsf{Adv}^{\text{MLWE}} +
\mathsf{Adv}^{\text{MLWE/MSIS}} + \mathsf{Adv}^{\text{KEA}}
= \mathsf{negl}(\lambda)
\]
Since all four are negligible by assumption, the forgery advantage is
negligible.
\end{proof}

\subsection{Parallel Liveness}

\begin{theorem}[Parallel Liveness]
\label{thm:par-liveness}
If $f < n/3$ validators are Byzantine and the network is synchronous
(after GST with delay $\leq \Delta$), then for every valid block $B$
proposed by an honest proposer, a valid QuasarCert is produced within
time $\max(\Delta_{\mathrm{RT}}, \Delta_{\mathrm{BLS}},
\Delta_{\mathrm{ZK}}) + O(\Delta)$.  An adversary must stall all three
paths simultaneously to prevent finality.
\end{theorem}

\begin{proof}
The three paths start upon block proposal and execute concurrently
(Section~\ref{sec:per-block}).  We show each completes independently.

\medskip
\textbf{BLS path.}
Each honest validator signs $H(B)$ upon receiving $B$.  After GST, $B$
reaches all honest validators within $\Delta$.  There are
$n - f > 2n/3 \geq t$ honest validators.  The proposer collects
$\geq t$ BLS signatures within $2\Delta$.  Aggregation is
non-interactive (${\sim}3$\,ms GPU).  Total: $2\Delta + O(1)$.

\medskip
\textbf{Corona path.}
Corona is a 2-round protocol.  Round~1 commitments arrive within
$\Delta$ of block receipt.  Round~2 responses arrive within another
$\Delta$ after $\geq t$ valid Round~1 messages are collected.  The
combiner assembles $\sigma_{\mathrm{RT}}$ in 7\,ms.  Total:
$2\Delta + O(1)$.  Byzantine validators may send invalid or absent
shares; with $\geq t$ honest participants the combiner always has
enough valid shares.  Invalid shares are detected by MAC authentication
and discarded.

\medskip
\textbf{Z-Chain path.}
Each honest validator signs $H(B)$ with ML-DSA-65 (measured
${\sim}495$\,$\mu$s sign, ${\sim}181$\,$\mu$s verify; Apple~M1~Max CPU).
Z-Chain collects honest signatures within $\Delta + O(1)$ and runs the
Groth16 prover over the ML-DSA-65 verification circuit
($\approx 2^{22.5}$ R1CS constraints; App.~B of
\texttt{proofs/quasar-cert-soundness.tex}).  Prover wall-clock is
${\sim}400$\,ms CPU or ${\sim}5$--$15$\,ms GPU; the 192-byte proof size
is circuit-independent.  For missing signatures from Byzantine validators,
Z-Chain waits up to a timeout $T_{\mathrm{ZK}}$, then produces a proof
for the $\geq t$ validators that did sign.
Total: $\Delta + T_{\mathrm{ZK}} + O(1)$.

\medskip
\textbf{Why all three must be stalled.}
Stalling any single path requires preventing a quorum of honest
validators from contributing.  The BLS quorum requires $t$ signers;
preventing it requires corrupting $> n - t \geq n/3$ validators.
The same bound applies to Corona and the ML-DSA collection.  Since
$f < n/3$, no bounded adversary can stall even one path, let alone all
three.
\end{proof}

\subsection{Post-Quantum Safety}

\begin{theorem}[Post-Quantum Safety]
\label{thm:pq-safety}
If the BLS12-381 co-CDH assumption is broken by a quantum adversary
(Shor's algorithm on pairings), QuasarCert verification restricted to
the Corona and ML-DSA components still guarantees soundness under the
Module-LWE and MLWE/MSIS assumptions.

\medskip\noindent\textit{Note: Corona's concrete quantum security
($\geq 2^{130}$) is derived from the BDGL sieving heuristic with
Grover speedup in Appendix~E of~\cite{qcert-proof}.  The Groth16
trusted-setup assumption is addressed in Appendix~D
of~\cite{qcert-proof}.}
\end{theorem}

\begin{proof}
Suppose a QPT adversary $\mathcal{A}_Q$ can forge BLS signatures.
Then $\mathcal{A}_Q$ can produce a valid $\sigma_{\mathrm{BLS}}$
without the honest quorum.  However, $\mathcal{A}_Q$ must also forge
$\sigma_{\mathrm{RT}}$ and $\pi_{\mathrm{ZK}}$.

\medskip
\textbf{Corona under quantum attack.}
Corona's security reduces to Module-LWE.  The best known quantum
attack on Module-LWE is lattice sieving with Grover speedup, achieving
$2^{0.265\beta}$ time for BKZ blocksize $\beta$.  With Corona
parameters ($N = 256$, $q \approx 2^{48}$, module dimension
$8 \times 7$, LWE dimension 2048), the quantum attack cost is
$\geq 2^{130}$ operations~\cite{corona}.  This exceeds $2^{\lambda}$
for $\lambda = 128$.  $\mathcal{A}_Q$ cannot forge
$\sigma_{\mathrm{RT}}$.

\medskip
\textbf{ML-DSA under quantum attack.}
ML-DSA-65 is parameterized for NIST Security Level~3: $\geq 128$-bit
quantum security.  The NIST analysis~\cite{nist-pqc} establishes that
the best quantum attack (Core-SVP with quantum sieving) requires
$\geq 2^{128}$ operations.  $\mathcal{A}_Q$ cannot forge an ML-DSA
signature, so it cannot produce a valid witness for the Groth16 circuit.

\medskip
\textbf{Groth16 under quantum attack.}
A quantum adversary can break the KEA assumption on the pairing group
(Shor extracts discrete logs, enabling proof forgery).  In this case,
Quasar falls back to direct ML-DSA verification: the verifier checks
$\geq t$ raw ML-DSA signatures instead of the Groth16 proof.  This is
a pre-configured fallback mode (Section~\ref{sec:implementation}).

\medskip
\textbf{Explicit reduction.}
Define $\mathsf{VerifyPQ}$ as the verification predicate restricted to
Corona and ML-DSA:
\[
  \mathsf{VerifyPQ}(B, \mathcal{V}, \mathcal{C}) =
  \mathsf{RT.Verify}(\cdot) \;\land\;
  \mathsf{MLDSA.VerifyQuorum}(\cdot)
\]
where $\mathsf{MLDSA.VerifyQuorum}$ checks either the Groth16 proof
(if pairing is trusted) or $\geq t$ raw ML-DSA signatures (fallback).

Suppose $\mathcal{A}_Q$ forges a certificate under
$\mathsf{VerifyPQ}$.  We construct a reduction $\mathcal{R}$:
\begin{enumerate}[nosep]
  \item $\mathcal{R}$ receives a Module-LWE challenge
    $(\mathbf{A}, \mathbf{b})$.
  \item $\mathcal{R}$ embeds $(\mathbf{A}, \mathbf{b})$ as the
    Corona public key and simulates the consensus protocol for
    $\mathcal{A}_Q$.
  \item If $\mathcal{A}_Q$ forges $\sigma_{\mathrm{RT}}$,
    $\mathcal{R}$ extracts a short vector solving the Module-SIS
    instance dual to the LWE challenge.
  \item If $\mathcal{A}_Q$ instead forges the ML-DSA component,
    $\mathcal{R}$ is constructed symmetrically as a reduction from
    MLWE/MSIS.
\end{enumerate}
By a standard hybrid argument (guess which component is forged, losing
a factor of 2):
\[
  \mathsf{Adv}^{\mathsf{VerifyPQ}}_{\mathcal{A}_Q} \leq
  2 \cdot \max(\mathsf{Adv}^{\text{MLWE}},
  \mathsf{Adv}^{\text{MLWE/MSIS}}) = \mathsf{negl}(\lambda)
\]
\end{proof}
