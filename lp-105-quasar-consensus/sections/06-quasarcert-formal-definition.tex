\section{QuasarCert: Formal Definition}

\label{sec:quasarcert}
%% ============================================================================

\begin{definition}[QuasarCert]
\label{def:quasarcert}
A QuasarCert for block $B$ with respect to validator set
$\mathcal{V} = \{v_1, \ldots, v_n\}$ is a triple
$\mathcal{C} = (\sigma_{\mathrm{RT}}, \sigma_{\mathrm{BLS}},
\pi_{\mathrm{ZK}})$ where:
\begin{enumerate}[nosep]
  \item $\sigma_{\mathrm{RT}}$ is a Corona threshold signature on
    $H(B)$ under the Q-Chain threshold public key
    $\mathit{pk}_{\mathrm{RT}}$, valid if
    $\mathsf{RT.Verify}(\mathit{pk}_{\mathrm{RT}}, H(B),
    \sigma_{\mathrm{RT}}) = 1$.
    Size: ${\sim}56$\,KB.
  \item $\sigma_{\mathrm{BLS}} \in \mathbb{G}_1$ is an aggregate
    BLS12-381 signature on $H(B)$, valid if
    $e(\sigma_{\mathrm{BLS}}, g_2) = e(H_1(B),
    \mathit{apk}_{\mathrm{BLS}})$ where $\mathit{apk}_{\mathrm{BLS}}$
    is the aggregate public key of the signing set.
    Size: 48 bytes (compressed $\mathbb{G}_1$).
  \item $\pi_{\mathrm{ZK}} \in \{0,1\}^{192}$ is a Groth16
    proof~\cite{groth2016} attesting:
    \[
      \exists\, (\sigma_1^{\mathrm{ML}}, \ldots,
      \sigma_n^{\mathrm{ML}}) \;\text{s.t.}\;
      \forall i \in [n]:\;
      \mathsf{MLDSA.Verify}(\mathit{pk}_i^{\mathrm{ML}}, H(B),
      \sigma_i^{\mathrm{ML}}) = 1
    \]
    Verified by
    $\mathsf{Groth16.Verify}(\mathit{vk}, x, \pi_{\mathrm{ZK}}) = 1$
    where $x = (H(B), \{\mathit{pk}_i^{\mathrm{ML}}\}_{i=1}^n)$.
    Size: 192 bytes (two $\mathbb{G}_1$ + one $\mathbb{G}_2$ point).
\end{enumerate}
\end{definition}

\begin{definition}[QuasarCert Verification]
\label{def:verify-qcert}
$\mathsf{VerifyQuasarCert}(B, \mathcal{V}, \mathcal{C})$ returns 1 iff:
\begin{align}
  &\mathsf{RT.Verify}(\mathit{pk}_{\mathrm{RT}}, H(B),
    \sigma_{\mathrm{RT}}) = 1 \label{eq:v-rt} \\
  &e(\sigma_{\mathrm{BLS}}, g_2) =
    e(H_1(B), \mathit{apk}_{\mathrm{BLS}}) \label{eq:v-bls} \\
  &\mathsf{Groth16.Verify}(\mathit{vk}, (H(B),
    \{\mathit{pk}_i^{\mathrm{ML}}\}), \pi_{\mathrm{ZK}}) = 1
    \label{eq:v-zk}
\end{align}
\end{definition}

\begin{remark}
In epoch mode (one Corona signature per epoch of $k$ blocks, with
per-block BLS only), the amortized overhead is $56\text{\,KB}/k + 48$\,B
per block.  With $k = 1000$ (1-second epochs at 1\,ms blocks), this is
${\sim}104$ bytes/block.
\end{remark}

\paragraph{Cryptographic assumptions.}
The QuasarCert relies on four assumptions:
\begin{enumerate}[nosep]
  \item co-CDH on BLS12-381~\cite{boneh2001short}: given
    $(g, g^a, h) \in \mathbb{G}_1^2 \times \mathbb{G}_2$, computing
    $h^a$ is hard.
  \item Module-LWE~\cite{regev2005}: distinguishing
    $(\mathbf{A}, \mathbf{As}+\mathbf{e})$ from
    $(\mathbf{A}, \mathbf{u})$ is hard (Corona basis).
  \item MLWE/MSIS~\cite{nist-pqc}: ML-DSA-65 (FIPS 204) security
    reduces to Module-LWE and Module-SIS with ${\geq}128$-bit quantum
    security (NIST Level~3).
  \item Knowledge of Exponent + Algebraic Group
    Model~\cite{groth2016}: Groth16 is knowledge-sound; a valid proof
    implies the prover knows a witness.
\end{enumerate}
