\section{System Model}

\label{sec:model}
%% ============================================================================

\begin{definition}[Network]
A set $\mathcal{V}$ of $n$ validators, of which at most $f < n/3$ are
Byzantine. Communication is point-to-point authenticated. The network is
partially synchronous: there exists an unknown Global Stabilization Time
(GST) after which all messages between correct validators are delivered within
a known bound $\Delta$.
\end{definition}

\begin{definition}[Cryptographic Configuration]
Each validator $v_i \in \mathcal{V}$ holds:
\begin{itemize}[leftmargin=2em]
\item A BLS12-381 secret key $\mathit{sk}_i^{\mathrm{BLS}}$ with public key
      $\mathit{pk}_i^{\mathrm{BLS}}$.
\item A share $\mathit{sk}_i^{\mathrm{RT}}$ of the Corona threshold key
      with public key $\mathit{pk}^{\mathrm{RT}}$ (threshold $t = \lceil 2n/3
      \rceil + 1$).
\item An ML-DSA-65 secret key $\mathit{sk}_i^{\mathrm{ML}}$ with public key
      $\mathit{pk}_i^{\mathrm{ML}}$.
\end{itemize}
\end{definition}

\begin{definition}[Consensus Modes]
\label{def:modes}
Quasar supports four operational modes, selectable at network genesis:
\begin{enumerate}[leftmargin=2em]
\item \textbf{BLS-only}: Classical aggregate signatures. Fastest finality.
      No post-quantum guarantees.
\item \textbf{BLS + ML-DSA}: Dual consensus. Post-quantum identity proof per
      validator. Certificates carry both a BLS aggregate and $\lceil 2n/3
      \rceil + 1$ ML-DSA-65 signatures.
\item \textbf{BLS + Corona}: Dual consensus. Post-quantum threshold proof.
      Certificates carry a BLS aggregate and a single Corona threshold
      signature.
\item \textbf{BLS + Corona + ML-DSA} (Quantum): All three layers. This is
      the mode analyzed in this paper.
\end{enumerate}
\end{definition}

The function \texttt{IsQuantumMode()} returns true when all three signing paths
are configured. The entry point \texttt{QuantumSignRound1} dispatches BLS,
Corona, and ML-DSA signing concurrently via goroutines.
