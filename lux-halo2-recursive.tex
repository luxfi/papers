\begin{abstract}
We present LP-3663, a native precompiled contract for Halo2 proof verification on the Lux EVM, with special emphasis on recursive proof composition through accumulation schemes. Halo2 achieves trustless setup by using Inner Product Arguments (IPA) instead of KZG commitments, enabling fully transparent zero-knowledge proofs. Our implementation at address \texttt{0x0904} supports both standard verification and recursive accumulation, where multiple proofs are folded into a single accumulator that is verified once. We detail the Pasta curves (Pallas and Vesta) for efficient recursion, present the accumulation protocol, and analyze the trade-offs between IPA-based and pairing-based approaches. The precompile achieves verification in approximately 350,000 gas for standard proofs and 180,000 gas for accumulated proofs, enabling practical recursive SNARKs on-chain.
\end{abstract}

\section{Introduction}

Halo2~\cite{halo2} represents a paradigm shift in zero-knowledge proof systems by eliminating the trusted setup requirement while maintaining practical verification efficiency. The key innovations are:

\begin{itemize}
    \item \textbf{Inner Product Arguments (IPA)}: Polynomial commitments based on discrete log, no pairings required
    \item \textbf{Accumulation Scheme}: Deferred verification through proof folding
    \item \textbf{Recursive Composition}: Proofs that verify other proofs within circuits
    \item \textbf{Pasta Curves}: Cycle of curves enabling efficient recursion
\end{itemize}

The Lux Network benefits from Halo2 for:
\begin{itemize}
    \item \textbf{Trustless rollups}: No ceremony required for new circuit deployments
    \item \textbf{Incremental computation}: Fold many state transitions into one proof
    \item \textbf{Post-quantum migration path}: IPA provides better PQ resistance than pairings
    \item \textbf{Privacy protocols}: Complex multi-step proofs composed recursively
\end{itemize}

\section{Inner Product Arguments}

\subsection{Commitment Scheme}

IPA commitments use a vector of independent generators:

\begin{definition}[IPA Setup]
    Generate random generators $\mathbf{G} = (G_1, \ldots, G_n) \in \mathbb{G}^n$ and blinding generator $H \in \mathbb{G}$.
\end{definition}

\begin{definition}[Pedersen Vector Commitment]
    For vector $\mathbf{a} = (a_1, \ldots, a_n) \in \mathbb{F}^n$ and blinding $r \in \mathbb{F}$:
    \[
    C = \langle \mathbf{a}, \mathbf{G} \rangle + r \cdot H = \sum_{i=1}^{n} a_i G_i + r H
    \]
\end{definition}

\subsection{Polynomial Commitment via IPA}

To commit to polynomial $p(X) = \sum_{i=0}^{n-1} a_i X^i$:

\begin{enumerate}
    \item Interpret coefficients as vector $\mathbf{a} = (a_0, \ldots, a_{n-1})$
    \item Compute commitment $C = \langle \mathbf{a}, \mathbf{G} \rangle$
\end{enumerate}

\subsection{Opening Protocol}

To prove $p(z) = v$ for committed polynomial:

\begin{algorithm}
\caption{IPA Opening Protocol}
\begin{algorithmic}[1]
\State \textbf{Prover input:} $C, \mathbf{a}, z, v$ where $p(z) = v$
\State \textbf{Verifier input:} $C, z, v$

\State Compute $\mathbf{b} = (1, z, z^2, \ldots, z^{n-1})$ \Comment{Lagrange basis at $z$}

\For{$j = \log_2(n)$ down to $1$}
    \State Split: $\mathbf{a} = (\mathbf{a}_L, \mathbf{a}_R)$, $\mathbf{b} = (\mathbf{b}_L, \mathbf{b}_R)$, $\mathbf{G} = (\mathbf{G}_L, \mathbf{G}_R)$

    \State Prover computes:
    \State \quad $L_j = \langle \mathbf{a}_L, \mathbf{G}_R \rangle + \langle \mathbf{a}_L, \mathbf{b}_R \rangle \cdot U$
    \State \quad $R_j = \langle \mathbf{a}_R, \mathbf{G}_L \rangle + \langle \mathbf{a}_R, \mathbf{b}_L \rangle \cdot U$

    \State Prover sends $(L_j, R_j)$
    \State Verifier sends challenge $u_j$

    \State Fold:
    \State \quad $\mathbf{a}' = \mathbf{a}_L + u_j \cdot \mathbf{a}_R$
    \State \quad $\mathbf{b}' = \mathbf{b}_L + u_j^{-1} \cdot \mathbf{b}_R$
    \State \quad $\mathbf{G}' = \mathbf{G}_L + u_j^{-1} \cdot \mathbf{G}_R$
\EndFor

\State Prover sends final $a' \in \mathbb{F}$

\State \textbf{Verification:}
\State $C' = C + \sum_{j} (u_j^2 L_j + u_j^{-2} R_j)$
\State Check: $C' = a' \cdot G' + (a' \cdot b') \cdot U$
\end{algorithmic}
\end{algorithm}

\begin{theorem}[IPA Soundness]
    Under the discrete logarithm assumption, a PPT adversary cannot produce an accepting opening $(z, v, \pi)$ for $C$ if $p(z) \neq v$, except with negligible probability.
\end{theorem}

\section{Pasta Curves}

\subsection{Curve Cycle}

Halo2 uses the Pasta curves---Pallas and Vesta---forming a 2-cycle:

\begin{definition}[Pallas Curve]
    \begin{align}
        E_p: y^2 &= x^3 + 5 \text{ over } \mathbb{F}_p \\
        p &= 2^{254} + 45560315531419706090280762371685220353 \\
        |\mathbb{G}_p| &= q \text{ (scalar field)}
    \end{align}
\end{definition}

\begin{definition}[Vesta Curve]
    \begin{align}
        E_q: y^2 &= x^3 + 5 \text{ over } \mathbb{F}_q \\
        q &= 2^{254} + 45560315531506369815346746415080538113 \\
        |\mathbb{G}_q| &= p \text{ (scalar field)}
    \end{align}
\end{definition}

\begin{proposition}[Cycle Property]
    The base field of Pallas equals the scalar field of Vesta, and vice versa. This enables:
    \begin{enumerate}
        \item Circuit on Pallas can efficiently verify Vesta proof (field elements match)
        \item Circuit on Vesta can efficiently verify Pallas proof
        \item Recursive composition alternates between curves
    \end{enumerate}
\end{proposition}

\section{Accumulation Scheme}

\subsection{Motivation}

Direct IPA verification requires $O(n)$ verifier time, too expensive for on-chain use. The accumulation scheme defers expensive verification:

\begin{definition}[Accumulator]
    An accumulator $acc$ represents deferred verification work. Multiple accumulators can be combined:
    \[
    acc_{new} = \text{Accumulate}(acc_1, acc_2, \ldots, acc_k)
    \]
    Final verification checks only the combined accumulator.
\end{definition}

\subsection{Protocol Overview}

\begin{algorithm}
\caption{Accumulation-Based Recursive Verification}
\begin{algorithmic}[1]
\State \textbf{Circuit $C$ on Pallas:}
\State \quad Receives $(\pi_{vesta}, acc_{old})$
\State \quad Verifies $\pi_{vesta}$ (produces new accumulator $acc_{vesta}$)
\State \quad Combines: $acc_{new} = \text{Fold}(acc_{old}, acc_{vesta})$
\State \quad Outputs: $(acc_{new}, \text{public\_output})$

\State \textbf{Prover generates:}
\State \quad Halo2 proof $\pi_{pallas}$ for circuit $C$

\State \textbf{Final verification (on-chain):}
\State \quad Verify $\pi_{pallas}$ (produces $acc_{final}$)
\State \quad Check $acc_{final}$ is valid (single IPA check)
\end{algorithmic}
\end{algorithm}

\subsection{Accumulator Structure}

\begin{definition}[IPA Accumulator]
    An accumulator contains:
    \[
    acc = (C, z, v, \mathbf{u}, G')
    \]
    where:
    \begin{itemize}
        \item $C$ is the commitment
        \item $z, v$ are the claimed evaluation point and value
        \item $\mathbf{u} = (u_1, \ldots, u_{\log n})$ are folding challenges
        \item $G'$ is the folded generator
    \end{itemize}
\end{definition}

\subsection{Accumulator Folding}

\begin{algorithm}
\caption{Accumulator Folding}
\begin{algorithmic}[1]
\Function{FoldAccumulators}{$acc_1, acc_2$}
    \State Parse $acc_1 = (C_1, z_1, v_1, \mathbf{u}_1, G'_1)$
    \State Parse $acc_2 = (C_2, z_2, v_2, \mathbf{u}_2, G'_2)$

    \State $r \gets H(acc_1, acc_2)$ \Comment{Fiat-Shamir challenge}

    \State $C_{fold} = C_1 + r \cdot C_2$
    \State $z_{fold} = z_1$ \Comment{Must match for folding}
    \State $v_{fold} = v_1 + r \cdot v_2$
    \State $\mathbf{u}_{fold} = \mathbf{u}_1 \oplus r \cdot \mathbf{u}_2$ \Comment{Element-wise}
    \State $G'_{fold} = G'_1 + r \cdot G'_2$

    \State \Return $(C_{fold}, z_{fold}, v_{fold}, \mathbf{u}_{fold}, G'_{fold})$
\EndFunction
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Folding Soundness]
    If both $acc_1$ and $acc_2$ are valid accumulators (i.e., would pass full IPA verification), then $acc_{fold}$ is also valid, except with probability $1/|\mathbb{F}|$.
\end{theorem}

\subsection{Full Verification Cost}

For circuit with $2^k$ rows and $m$ advice columns:

\begin{equation}
G_{verify} = G_{base} + G_{IPA} + G_{lookups} + G_{permutation}
\end{equation}

\begin{align}
G_{base} &= 50,000 \\
G_{IPA} &= k \times 12,000 + m \times 3,000 \\
G_{lookups} &= \ell \times 5,000 \text{ (for $\ell$ lookup arguments)} \\
G_{permutation} &= 25,000
\end{align}

\begin{proposition}[Standard Circuit Cost]
    For a typical circuit with $k=17$ ($2^{17}$ rows), 5 advice columns, and 2 lookup arguments:
    \[
    G_{total} = 50,000 + 17 \times 12,000 + 5 \times 3,000 + 2 \times 5,000 + 25,000 = 349,000
    \]
\end{proposition}

\subsection{Accumulator Verification Cost}

Verifying only the final accumulator (deferred IPA check):

\begin{equation}
G_{acc} = 15,000 + k \times 5,000 + 50,000 \approx 150,000
\end{equation}

This is significantly cheaper than full proof verification, making recursive composition practical.

\subsection{Accumulator Chip}

\begin{lstlisting}[language=Rust, caption={Accumulator Folding Chip}]
struct AccumulatorChip<C: CurveAffine> {
    config: AccumulatorConfig,
    _marker: PhantomData<C>,
}

impl<C: CurveAffine> AccumulatorChip<C> {
    fn fold(
        &self,
        region: &mut Region<C::Scalar>,
        acc1: Accumulator<C>,
        acc2: Accumulator<C>,
    ) -> Result<Accumulator<C>, Error> {
        // Generate random linear combination challenge
        let r = self.random_challenge(region, &acc1, &acc2)?;

        // Fold commitments: C = C1 + r * C2
        let c_fold = self.point_add_scalar(
            region,
            acc1.commitment,
            acc2.commitment,
            r,
        )?;

        // Fold values: v = v1 + r * v2
        let v_fold = self.field_add_mul(
            region,
            acc1.v,
            acc2.v,
            r,
        )?;

        Ok(Accumulator {
            commitment: c_fold,
            z: acc1.z, // Must match
            v: v_fold,
            challenges: fold_challenges(acc1.challenges, acc2.challenges, r),
            g_final: fold_generators(acc1.g_final, acc2.g_final, r),
        })
    }
}
\end{lstlisting}

\subsection{Precompile Entry Point}

\begin{lstlisting}[language=Go, caption={Halo2 Precompile Implementation}]
func (p *Halo2Precompile) Run(
    accessibleState contract.AccessibleState,
    caller common.Address,
    addr common.Address,
    input []byte,
    suppliedGas uint64,

    default:

    }
}

\subsection{IPA Verification}

\begin{lstlisting}[language=Go, caption={IPA Opening Verification}]
func (p *Halo2Precompile) ipaOpenVerify(
    input []byte,
    curve uint8,
    suppliedGas uint64,
) ([]byte, uint64, error) {
    // Parse input
    commitment, z, v, proof, err := parseIPAInput(input)
    if err != nil {
        return nil, suppliedGas, err
    }

    // Calculate gas
    numRounds := len(proof.L)
    gasCost := 10000 + uint64(numRounds)*200

    if suppliedGas < gasCost {
        return nil, 0, ErrOutOfGas
    }

    // Select curve
    var valid bool
    if curve == 0 {
        valid = verifyIPAPallas(commitment, z, v, proof)
    } else {
        valid = verifyIPAVesta(commitment, z, v, proof)
    }

    result := make([]byte, 32)
    if valid {
        result[31] = 1
    }
    return result, suppliedGas - gasCost, nil
}
\end{lstlisting}

\subsection{MSM Optimization}

The verifier's main cost is multi-scalar multiplication. We optimize using Pippenger's algorithm:

\begin{lstlisting}[language=Go, caption={Optimized MSM}]
func pippenger(
    points []PastaPoint,
    scalars []PastaScalar,
) PastaPoint {
    // Window size based on input size
    windowSize := optimalWindowSize(len(points))

    // Build buckets
    numBuckets := 1 << windowSize
    buckets := make([]PastaPoint, numBuckets)

    // Process each window
    numWindows := (254 + windowSize - 1) / windowSize
    result := PastaPoint{}

    for w := numWindows - 1; w >= 0; w-- {
        // Double result windowSize times
        for i := 0; i < windowSize; i++ {
            result = result.Double()
        }

        // Clear buckets
        for i := range buckets {
            buckets[i] = PastaPoint{}
        }

        // Add points to buckets
        for i, scalar := range scalars {
            bucket := extractWindow(scalar, w, windowSize)
            if bucket > 0 {
                buckets[bucket] = buckets[bucket].Add(points[i])
            }
        }

        // Aggregate buckets
        result = result.Add(aggregateBuckets(buckets))
    }

    return result
}
\end{lstlisting}

\section{Security Analysis}

\subsection{Assumptions}

\begin{enumerate}
    \item \textbf{Discrete Logarithm}: Computing $\log_G H$ is hard for random $G, H$
    \item \textbf{Random Oracle Model}: Hash functions behave as random oracles
\end{enumerate}

\begin{theorem}[Halo2 Security]
    Under the discrete logarithm assumption and random oracle model, Halo2 is:
    \begin{enumerate}
        \item \textbf{Complete}: Valid proofs always verify
        \item \textbf{Sound}: Invalid proofs are rejected with overwhelming probability
        \item \textbf{Zero-Knowledge}: Proofs reveal nothing beyond statement validity
    \end{enumerate}
\end{theorem}

\subsection{Accumulator Security}

\begin{lemma}[Accumulator Binding]
    An adversary cannot produce two different $(z, v)$ pairs for the same accumulator, except with negligible probability under DLP.
\end{lemma}

\begin{lemma}[Folding Security]
    If either input accumulator is invalid, the folded accumulator is invalid with probability $1 - 1/|\mathbb{F}|$.
\end{lemma}

\section{Applications}

\subsection{Incrementally Verifiable Computation}

\begin{lstlisting}[language=Rust, caption={IVC Circuit Structure}]
struct IVCCircuit<F: Field> {
    // Previous state
    prev_state: AssignedCell<F>,
    prev_accumulator: Accumulator,

    // Transition
    input: AssignedCell<F>,

    // New state
    new_state: AssignedCell<F>,
}

impl<F: Field> Circuit<F> for IVCCircuit<F> {
    fn synthesize(&self, cs: &mut ConstraintSystem<F>) {
        // 1. Verify previous proof's accumulator
        let prev_acc = verify_accumulator(cs, &self.prev_accumulator);

        // 2. Apply state transition
        let new_state = transition_function(cs, self.prev_state, self.input);
        cs.enforce_equal(new_state, self.new_state);

        // 3. Fold accumulators
        let circuit_acc = self.current_accumulator();
        let folded = fold_accumulators(cs, prev_acc, circuit_acc);

        // 4. Output new accumulator
        expose_public(cs, folded);
    }
}
\end{lstlisting}

\section{Accumulator Serialization}

\begin{lstlisting}[language=Go, caption={Accumulator Encoding}]
func (acc *Accumulator) Encode() []byte {
    var buf bytes.Buffer

    // Commitment (64 bytes)
    buf.Write(acc.Commitment.Encode())

    // Evaluation point (32 bytes)
    buf.Write(acc.Z.Bytes())

    // Claimed value (32 bytes)
    buf.Write(acc.V.Bytes())

    // Number of challenges (4 bytes)
    binary.BigEndian.PutUint32(buf.Next(4), uint32(len(acc.Challenges)))

    // Challenges (32 bytes each)
    for _, c := range acc.Challenges {
        buf.Write(c.Bytes())
    }

    // Final generator (64 bytes)
    buf.Write(acc.GFinal.Encode())

    return buf.Bytes()
}
\end{lstlisting}
