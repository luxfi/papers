\section{Privacy--Compliance Tradeoff}
\label{sec:tradeoff}

\subsection{Formal Tradeoff}

\begin{definition}[Privacy--Compliance Score]
For association set $\mathcal{A}$:
\begin{align}
    \text{Privacy}(\mathcal{A}) &= \log_2(k_{\text{eff}}) \quad \text{(bits of anonymity)} \\
    \text{Compliance}(\mathcal{A}) &= 1 - \frac{|\mathcal{A} \cap \mathcal{S}|}{|\mathcal{A}|} \quad \text{(fraction non-sanctioned)}
\end{align}
where $\mathcal{S}$ is the set of sanctioned deposits.
\end{definition}

\begin{theorem}[Tradeoff Bound]
For a properly maintained association set (no sanctioned deposits):
\begin{equation}
    \text{Privacy}(\mathcal{A}) + \text{Compliance}(\mathcal{A}) \geq \log_2(|\mathcal{A}|) + 1
\end{equation}
Larger sets provide both better privacy and perfect compliance.
\end{theorem}

\begin{proof}
If $\mathcal{A} \cap \mathcal{S} = \emptyset$ (proper maintenance), then $\text{Compliance} = 1$. Privacy $= \log_2(k_{\text{eff}}) \leq \log_2(|\mathcal{A}|)$. The sum is $\geq \log_2(|\mathcal{A}|) + 1$.
\end{proof}
