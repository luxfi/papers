\section{Multi-Chain Privacy}
\label{sec:multichain}

\subsection{Cross-Appchain Private Transfers}

Lux Privacy Pools extend across appchains via Warp Messaging:

\begin{algorithm}[H]
\caption{Cross-Appchain Private Transfer}
\label{alg:cross_private}
\begin{algorithmic}[1]
\Require Deposit on appchain $A$, withdraw on appchain $B$
\State User deposits on appchain $A$ privacy pool (Algorithm~\ref{alg:deposit})
\State Appchain $A$ pool root $R_A$ is shared with appchain $B$ via Warp Message
\State Appchain $B$ privacy pool stores verified roots from other appchains
\State User generates ZK proof against $R_A$ on appchain $B$
\State Appchain $B$ contract verifies proof and Warp-verified root
\State Withdrawal is processed on appchain $B$
\State Nullifier is propagated back to appchain $A$ via Warp Message
\end{algorithmic}
\end{algorithm}

\subsection{Root Synchronization}

\begin{definition}[Cross-Chain Root Registry]
Each privacy pool maintains a set of valid roots from connected appchains:
\begin{equation}
    \mathcal{R}_k = \{(R_j, h_j, \Pi_j) : j \in \text{connected appchains}\}
\end{equation}
where $R_j$ is appchain $j$'s pool root at height $h_j$, verified by Warp proof $\Pi_j$.
\end{definition}

\begin{proposition}[Cross-Chain Anonymity]
Cross-appchain privacy pools merge anonymity sets: a deposit on appchain $A$ and withdrawal on appchain $B$ achieves anonymity set $k_{\text{eff}} = k_A + k_B$ (deposits from both pools).
\end{proposition}
