\section{Decentralization Metrics}
\label{sec:decentralization}

\subsection{Nakamoto Coefficient}

\begin{definition}[Nakamoto Coefficient]
The Nakamoto coefficient $\mathsf{NC}$ is the minimum number of validators whose combined stake exceeds $W/3$:
\begin{equation}
    \mathsf{NC} = \min \left\{ k : \sum_{i=1}^{k} s_{\pi(i)}^{\text{eff}} \geq W/3 \right\}
\end{equation}
where $\pi$ sorts validators by descending effective stake.
\end{definition}

\begin{proposition}[NC Lower Bound Under Stake Cap]
With delegation multiplier $\omega$, $\mathsf{NC} \geq \lceil W / (3 \omega s^{\min}_{\text{val}}) \rceil$.
\end{proposition}

\subsection{Gini Coefficient of Stake Distribution}

\begin{definition}[Stake Gini]
\begin{equation}
    G = \frac{\sum_{i=1}^{n} \sum_{j=1}^{n} |s_i - s_j|}{2n \sum_{i=1}^{n} s_i}
\end{equation}
$G = 0$ indicates perfect equality; $G = 1$ indicates maximum concentration.
\end{definition}

Target: $G \leq 0.60$ for a healthy validator set. The delegation cap ensures $G$ cannot exceed $G_{\max} = 1 - 1/\omega \approx 0.93$.
