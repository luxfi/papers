\subsection{Slashing Execution}

\begin{algorithm}[H]
\caption{Slashing Execution Protocol}
\label{alg:slashing}
\begin{algorithmic}[1]
\Procedure{Slash}{validator $v_i$, violation type, evidence}
    \State Verify evidence cryptographically
    \State $p \gets \text{PenaltyRate}(\text{violation type})$
    \State $\text{slashAmount} \gets p \cdot s_i$
    \State Deduct from $v_i$'s self-stake first
    \If{$\text{slashAmount} > s_i^{\text{self}}$}
        \State Deduct remainder pro-rata from delegators
    \EndIf
    \State Burn 50\% of slashed amount
    \State Send 50\% to evidence submitter as bounty
    \State Jail $v_i$ for prescribed duration
    \If{$s_i^{\text{self}} < s^{\min}_{\text{val}}$ after slash}
        \State Force-exit $v_i$; begin unbonding for all delegators
    \EndIf
\EndProcedure
\end{algorithmic}
\end{algorithm}

\subsection{Equivocation Detection}

\begin{definition}[Equivocation]
Validator $v_i$ equivocates if it signs two conflicting blocks $B_1 \neq B_2$ at the same height $h$:
\begin{equation}
    \text{Equivocation}(v_i, h) \iff \exists\, B_1, B_2 : B_1 \neq B_2 \wedge \mathsf{Verify}(\mathsf{pk}_i, B_1, \sigma_1) \wedge \mathsf{Verify}(\mathsf{pk}_i, B_2, \sigma_2)
\end{equation}
\end{definition}

Equivocation is cryptographically provable: anyone holding both signed blocks can submit them as evidence. The proof is non-interactive and verifiable on-chain.

\subsection{Correlated Slashing}

When multiple validators misbehave simultaneously, the penalty is amplified:

\begin{definition}[Correlated Penalty]
If $k$ validators are slashed within the same epoch, each receives an additional penalty:
\begin{equation}\label{eq:correlated}
    p_{\text{correlated}} = p_{\text{base}} \cdot \min\left(3, 1 + \frac{\sum_{j \in \text{slashed}} w_j}{W/3}\right)
\end{equation}
\end{definition}

This graduated penalty structure deters coordinated attacks: by Equation~(\ref{eq:correlated}) the multiplier is $\min(3, 1 + \tfrac{\sum w_j}{W/3})$, so a coordinated equivocation involving $1/3$ of stake costs $2 \times 5\% = 10\%$ per validator (and involving $2/3$ of stake costs the capped $3 \times 5\% = 15\%$), making the attack economically irrational.
