\section{The Nova Protocol}

\label{sec:protocol}

\subsection{Frontier Sampling}

Nova samples the frontier collectively: in each round, the validator queries $k$ peers
about their preferences for \emph{all} frontier vertices simultaneously.

\begin{algorithm}[H]
\caption{Nova Protocol --- Frontier-Based Burst Finalization}
\label{alg:nova-core}
\begin{algorithmic}[1]
\Require Validator $v$, frontier $F$
\Require Parameters: $k$ (sample size), $\alpha$ (quorum), $\beta$ (threshold)
\While{$F \neq \emptyset$}
    \State $S \gets \text{RandomSample}(V \setminus \{v\}, k)$
    \State Query each $u \in S$ for preferences on all $w \in F$
    \For{each frontier vertex $w \in F$}
        \State Compute supermajority for $w$'s preferred conflict resolution
        \If{supermajority $\geq \alpha k$}
            \State $d[w] \gets d[w] + 1$
        \EndIf
        \If{$d[w] \geq \beta$}
            \State $B \gets \text{burst}(w)$ \Comment{collect unfinalized ancestors}
            \State \textbf{finalize all} $u \in B$ \Comment{supernova burst}
            \State Update frontier: $F \gets F \setminus \{w\} \cup \text{NewFrontier}(B)$
        \EndIf
    \EndFor
    \State Add new incoming vertices to $F$ as appropriate
\EndWhile
\end{algorithmic}
\end{algorithm}

\subsection{Conflict Resolution in Bursts}

\begin{algorithm}[H]
\caption{Nova Conflict Resolution During Burst}
\label{alg:nova-conflict}
\begin{algorithmic}[1]
\Function{ResolveBurst}{burst $B$}
    \State $\text{accepted} \gets \emptyset$
    \For{each $v \in \text{TopologicalSort}(B)$} \Comment{process in dependency order}
        \If{$\mathcal{C}(v) = \emptyset$ \textbf{or} $\mathcal{C}(v) \cap \text{accepted} = \emptyset$}
            \State $\text{accepted} \gets \text{accepted} \cup \{v\}$
        \Else
            \State \Comment{Conflict: prefer $v$ if it has higher confidence}
            \For{each $v' \in \mathcal{C}(v) \cap \text{accepted}$}
                \If{$d[v] > d[v']$}
                    \State $\text{accepted} \gets (\text{accepted} \setminus \{v'\}) \cup \{v\}$
                \EndIf
            \EndFor
        \EndIf
    \EndFor
    \Return $\text{accepted}$
\EndFunction
\end{algorithmic}
\end{algorithm}
