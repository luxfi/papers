\section{System Model}

\label{sec:model}

\begin{definition}[DAG Structure]
The transaction DAG $G = (V, E)$ consists of vertices $V$ (transactions) and directed
edges $E$ where $(u, v) \in E$ means $v$ is a parent of $u$. Each vertex $v$ has:
\begin{itemize}[nosep]
\item A set of parents $\text{parents}(v) \subseteq V$ with $|\text{parents}(v)| \geq 1$.
\item An ancestry cone $\text{cone}(v) = \{u \in V : u \text{ is reachable from } v\}$.
\item A conflict set $\mathcal{C}(v)$ of vertices consuming the same UTXO.
\end{itemize}
\end{definition}

\begin{definition}[Frontier]
The frontier $F(r)$ at round $r$ is the set of maximal unfinalized vertices:
\[
F(r) = \{v \in V : v \text{ is not finalized and } \nexists u \text{ unfinalized with } v \in \text{cone}(u)\}
\]
Intuitively, frontier vertices are the ``tips'' of the unfinalized DAG.
\end{definition}

\begin{definition}[Supernova Burst]
A supernova burst at round $r$ triggered by frontier vertex $v$ is the set:
\[
\text{burst}(v, r) = \{u \in \text{cone}(v) : u \text{ is not yet finalized}\}
\]
All vertices in the burst are finalized simultaneously.
\end{definition}
