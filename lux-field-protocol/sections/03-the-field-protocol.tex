\section{The Field Protocol}

\label{sec:protocol}

\subsection{Core Protocol}

\begin{algorithm}[H]
\caption{Field Protocol --- Multi-Valued Consensus}
\label{alg:field-core}
\begin{algorithmic}[1]
\Require Validator $i$, $d$-valued conflict with initial preference $\mathbf{p}_i$
\Require Parameters: $k$ (sample size), $\alpha$ (alignment threshold), $\beta$ (acceptance threshold), $\eta$ (update rate)
\State $\text{count}[c] \gets 0$ for all $c \in \{1, \ldots, d\}$
\While{$\max_c \text{count}[c] < \beta$}
    \State $S \gets \text{RandomSample}(V \setminus \{i\}, k)$
    \State Query each $j \in S$ for $\mathbf{p}_j$
    \State $\bar{\mathbf{p}} \gets \frac{1}{k} \sum_{j \in S} \mathbf{p}_j$ \Comment{sample mean preference}
    \State \Comment{Update preference vector toward sample mean}
    \State $\mathbf{p}_i \gets (1 - \eta) \mathbf{p}_i + \eta \bar{\mathbf{p}}$ \Comment{field alignment}
    \State $\mathbf{p}_i \gets \mathbf{p}_i / \|\mathbf{p}_i\|_1$ \Comment{renormalize to simplex}
    \State $c^* \gets \arg\max_c \bar{p}^{(c)}$ \Comment{dominant color in sample}
    \If{$\bar{p}^{(c^*)} \geq \alpha$} \Comment{supermajority for $c^*$}
        \State $\text{count}[c^*] \gets \text{count}[c^*] + 1$
    \EndIf
    \If{$\max_c \text{count}[c] \geq \beta$}
        \State \textbf{finalize} value $\arg\max_c \text{count}[c]$
    \EndIf
\EndWhile
\end{algorithmic}
\end{algorithm}

\subsection{Simplified Binary Mode}

When $d = 2$, the preference vector reduces to a scalar $p_i = p_i^{(1)}$ (with
$p_i^{(0)} = 1 - p_i$), and the protocol reduces to the Wave dynamics with continuous
preferences. The update becomes:
\[
p_i \gets (1 - \eta) p_i + \eta \bar{p}
\]
which converges to $p_i \in \{0, 1\}$ under the same conditions as Wave.
