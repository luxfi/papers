\begin{abstract}
We present a comprehensive mathematical framework for perpetual futures contracts on the Lux blockchain. Perpetual futures enable leveraged exposure to asset prices without expiration dates, using funding rate mechanisms to anchor contract prices to underlying spot markets. We derive optimal funding rate formulas, analyze liquidation engine design, and prove safety properties for position management. Our framework addresses key challenges including oracle manipulation resistance, cascading liquidation prevention, and capital efficiency. Implementation guidelines demonstrate Lux's suitability for high-frequency perpetual trading with sub-second finality.
\end{abstract}

\section{Introduction}

Perpetual futures contracts, pioneered by BitMEX in 2016, have become the dominant derivative instrument in cryptocurrency markets. Unlike traditional futures with fixed expiration dates, perpetuals use a funding rate mechanism to maintain price alignment with spot markets.

Key design challenges include:

\begin{enumerate}
\item \textbf{Price Anchoring}: Keeping perpetual price aligned with spot
\item \textbf{Leverage Management}: Enabling high leverage while maintaining solvency
\item \textbf{Liquidation Efficiency}: Resolving undercollateralized positions promptly
\item \textbf{Oracle Security}: Resistance to price manipulation attacks
\end{enumerate}

This paper provides rigorous mathematical foundations for each component.

\section{Perpetual Futures Mechanics}

\subsection{Position Definition}

\begin{definition}[Perpetual Position]
A perpetual position is defined by:
\begin{itemize}
\item $s \in \{-1, +1\}$: Side (short/long)
\item $q$: Position size (base asset units)
\item $P_e$: Entry price
\item $M$: Margin (collateral)
\item $L$: Leverage
\end{itemize}
\end{definition}

\begin{definition}[Notional Value]
Position notional value:
\begin{equation}
N = q \cdot P_e
\end{equation}
\end{definition}

\begin{definition}[Leverage]
\begin{equation}
L = \frac{N}{M} = \frac{q \cdot P_e}{M}
\end{equation}
\end{definition}

\subsection{Profit and Loss}

\begin{theorem}[Position PnL]
For mark price $P_m$ and entry price $P_e$:
\begin{equation}
\text{PnL} = s \cdot q \cdot (P_m - P_e)
\end{equation}
Long ($s = +1$) profits when price rises; short ($s = -1$) profits when price falls.
\end{theorem}

\begin{corollary}[Return on Margin]
The return on margin (RoM):
\begin{equation}
\text{RoM} = \frac{\text{PnL}}{M} = s \cdot L \cdot \frac{P_m - P_e}{P_e}
\end{equation}
Leverage amplifies returns proportionally.
\end{corollary}

\subsection{Margin and Equity}

\begin{definition}[Account Equity]
Total account value including unrealized PnL:
\begin{equation}
E = M + \text{PnL} = M + s \cdot q \cdot (P_m - P_e)
\end{equation}
\end{definition}

\begin{definition}[Margin Ratio]
\begin{equation}
\text{MR} = \frac{E}{N} = \frac{M + \text{PnL}}{q \cdot P_m}
\end{equation}
\end{definition}

\section{Funding Rate Mechanism}

\subsection{Purpose}

Funding rates serve to anchor the perpetual price $P_{\text{perp}}$ to the spot price $P_{\text{spot}}$:

\begin{itemize}
\item $P_{\text{perp}} > P_{\text{spot}}$: Longs pay shorts (reduces long demand)
\item $P_{\text{perp}} < P_{\text{spot}}$: Shorts pay longs (reduces short demand)
\end{itemize}

\subsection{Funding Rate Formula}

\begin{definition}[Premium Index]
The premium of perpetual over spot:
\begin{equation}
P_{\text{premium}} = \frac{P_{\text{perp}} - P_{\text{spot}}}{P_{\text{spot}}}
\end{equation}
\end{definition}

\begin{definition}[Funding Rate]
The funding rate combines premium with interest rate differential:
\begin{equation}
F = P_{\text{premium}} + \text{clamp}\left(\frac{r_{\text{quote}} - r_{\text{base}}}{n_{\text{funding}}}, -\delta, \delta\right)
\end{equation}
where:
\begin{itemize}
\item $r_{\text{quote}}, r_{\text{base}}$: Interest rates (typically both near 0.01\%/8h)
\item $n_{\text{funding}}$: Funding intervals per day (typically 3)
\item $\delta$: Clamp bound (e.g., 0.05\%)
\end{itemize}
\end{definition}

\begin{definition}[Time-Weighted Average Premium]
To resist manipulation:
\begin{equation}
P_{\text{premium}}^{\text{TWAP}} = \frac{1}{T} \int_{t-T}^{t} \frac{P_{\text{perp}}(\tau) - P_{\text{spot}}(\tau)}{P_{\text{spot}}(\tau)} d\tau
\end{equation}
Typical window $T = 8$ hours.
\end{definition}

\subsection{Funding Payment}

\begin{theorem}[Funding Payment]
At funding time, a position pays/receives:
\begin{equation}
\text{Funding Payment} = F \cdot N = F \cdot q \cdot P_m
\end{equation}
\begin{itemize}
\item $F > 0$: Longs pay, shorts receive
\item $F < 0$: Shorts pay, longs receive
\end{itemize}
\end{theorem}

\subsection{Equilibrium Analysis}

\begin{theorem}[Funding Rate Equilibrium]
At equilibrium, $P_{\text{perp}} = P_{\text{spot}}$:
\begin{equation}
F^* = \frac{r_{\text{quote}} - r_{\text{base}}}{n_{\text{funding}}}
\end{equation}
\end{theorem}

\begin{proof}
When $P_{\text{perp}} = P_{\text{spot}}$, premium is zero. The funding rate reduces to the interest rate differential. This creates a cost of carry analogous to traditional futures, where $F^*$ represents the equilibrium cost/benefit of holding leveraged exposure.
\end{proof}

\begin{proposition}[Convergence Property]
The funding mechanism is self-correcting:
\begin{equation}
\frac{dP_{\text{perp}}}{dt} = -k \cdot (P_{\text{perp}} - P_{\text{spot}})
\end{equation}
where $k > 0$ depends on trader responsiveness to funding costs.
\end{proposition}

\section{Mark Price and Index Price}

\subsection{Index Price}

\begin{definition}[Index Price]
Spot price aggregated from multiple sources:
\begin{equation}
P_{\text{index}} = \text{Median}(\{P_1, P_2, \ldots, P_n\})
\end{equation}
or weighted average with outlier filtering.
\end{definition}

\begin{algorithm}
\caption{Index Price Calculation}
\begin{algorithmic}[1]
\Require Price feeds $\{P_i\}_{i=1}^n$ with weights $\{w_i\}$
\Ensure Index price $P_{\text{index}}$
\State Remove feeds outside $\pm 5\%$ of median
\State $P_{\text{index}} \gets \sum_i w_i \cdot P_i / \sum_i w_i$
\State \Return $P_{\text{index}}$
\end{algorithmic}
\end{algorithm}

\subsection{Mark Price}

\begin{definition}[Mark Price]
Price used for PnL and liquidation calculations:
\begin{equation}
P_{\text{mark}} = P_{\text{index}} \cdot (1 + F_{\text{basis}})
\end{equation}
where $F_{\text{basis}}$ is the funding basis adjustment.
\end{definition}

Using mark price (rather than last traded price) for liquidations prevents manipulation via thin order book attacks.

\section{Liquidation Engine}

\subsection{Liquidation Trigger}

\begin{definition}[Maintenance Margin]
Minimum margin required:
\begin{equation}
M_{\text{maint}} = \text{MMR} \cdot N = \text{MMR} \cdot q \cdot P_m
\end{equation}
where MMR is the maintenance margin rate (e.g., 0.5\% to 2.5\%).
\end{definition}

\begin{definition}[Liquidation Condition]
A position is liquidatable when:
\begin{equation}
E < M_{\text{maint}}
\end{equation}
Equivalently:
\begin{equation}
\text{MR} < \text{MMR}
\end{equation}
\end{definition}

\begin{theorem}[Liquidation Price]
For a long position:
\begin{equation}
P_{\text{liq}} = P_e \cdot \frac{1 - \text{IMR}}{1 - \text{MMR}}
\end{equation}
For a short position:
\begin{equation}
P_{\text{liq}} = P_e \cdot \frac{1 + \text{IMR}}{1 + \text{MMR}}
\end{equation}
where IMR is initial margin rate ($= 1/L$).
\end{theorem}

\begin{proof}
Long position liquidates when:
\begin{align}
M + q(P_{\text{liq}} - P_e) &= \text{MMR} \cdot q \cdot P_{\text{liq}} \\
\frac{N}{L} + q \cdot P_{\text{liq}} - q \cdot P_e &= \text{MMR} \cdot q \cdot P_{\text{liq}} \\
\frac{P_e}{L} + P_{\text{liq}} - P_e &= \text{MMR} \cdot P_{\text{liq}} \\
P_{\text{liq}}(1 - \text{MMR}) &= P_e(1 - 1/L) = P_e(1 - \text{IMR}) \\
P_{\text{liq}} &= P_e \cdot \frac{1 - \text{IMR}}{1 - \text{MMR}}
\end{align}
\end{proof}

\subsection{Liquidation Process}

\begin{algorithm}
\caption{Position Liquidation}
\begin{algorithmic}[1]
\Require Position $P$, mark price $P_m$
\Ensure Position closed, insurance fund updated
\State \textbf{require} $E < M_{\text{maint}}$
\State $\text{remainingMargin} \gets M + \text{PnL}$
\State Close position at $P_m$ (take from order book or ADL)
\State $\text{liquidationFee} \gets \text{LiqFeeRate} \cdot N$
\If{$\text{remainingMargin} > \text{liquidationFee}$}
\State Add $\text{remainingMargin} - \text{liquidationFee}$ to insurance fund
\Else
\State Draw from insurance fund to cover shortfall
\EndIf
\State Pay liquidator fee
\end{algorithmic}
\end{algorithm}

\subsection{Insurance Fund}

\begin{definition}[Insurance Fund]
Reserve to cover liquidation shortfalls:
\begin{equation}
\text{IF}_t = \text{IF}_{t-1} + \sum_{\text{liquidations}} (\text{remainingMargin} - \text{fee}) - \sum_{\text{shortfalls}}
\end{equation}
\end{definition}

\begin{proposition}[Insurance Fund Sizing]
Target insurance fund size based on:
\begin{equation}
\text{IF}_{\text{target}} \geq \text{VaR}_{99\%}(\text{OpenInterest}, \text{Volatility}, \text{Leverage})
\end{equation}
\end{proposition}

\subsection{Auto-Deleveraging (ADL)}

When insurance fund is insufficient:

\begin{definition}[ADL Priority]
Positions ranked for deleveraging by:
\begin{equation}
\text{ADL Rank} = \text{PnL Percentile} \cdot \text{Leverage Percentile}
\end{equation}
Highest-profit, highest-leverage positions deleveraged first.
\end{definition}

\begin{algorithm}
\caption{Auto-Deleveraging}
\begin{algorithmic}[1]
\Require Bankrupt position $B$, shortfall $S$
\Ensure Shortfall covered via ADL
\State Sort opposing positions by ADL rank (descending)
\For{each position $P$ until $S \leq 0$}
\State $\text{deleverageQty} \gets \min(P.\text{qty}, S / P_m)$
\State Close $\text{deleverageQty}$ from $P$ against $B$
\State $S \gets S - \text{deleverageQty} \cdot P_m$
\EndFor
\end{algorithmic}
\end{algorithm}

\section{Position Sizing and Risk}

\subsection{Maximum Leverage}

\begin{definition}[Leverage Tiers]
Larger positions require lower leverage:
\begin{equation}
L_{\max}(N) = \begin{cases}
100\times & N \leq \$50\text{k} \\
50\times & N \leq \$250\text{k} \\
20\times & N \leq \$1\text{M} \\
10\times & N > \$1\text{M}
\end{cases}
\end{equation}
\end{definition}

\begin{proposition}[Tier Rationale]
Lower leverage for larger positions because:
\begin{enumerate}
\item Larger liquidations have more market impact
\item Slippage increases with order size
\item Reduces systemic risk from whale positions
\end{enumerate}
\end{proposition}

\subsection{Open Interest Limits}

\begin{definition}[Open Interest Cap]
Maximum total long/short positions:
\begin{equation}
\text{OI}_{\max} = f(\text{SpotLiquidity}, \text{OracleDepth})
\end{equation}
\end{definition}

\section{Order Types}

\subsection{Limit Orders}

\begin{definition}[Limit Order]
Order to buy/sell at specified price or better:
\begin{equation}
\text{Execute if} \begin{cases}
P_{\text{market}} \leq P_{\text{limit}} & \text{(buy)} \\
P_{\text{market}} \geq P_{\text{limit}} & \text{(sell)}
\end{cases}
\end{equation}
\end{definition}

\subsection{Stop Orders}

\begin{definition}[Stop-Loss Order]
Triggered when mark price crosses threshold:
\begin{equation}
\text{Trigger if} \begin{cases}
P_m \leq P_{\text{stop}} & \text{(sell stop)} \\
P_m \geq P_{\text{stop}} & \text{(buy stop)}
\end{cases}
\end{equation}
\end{definition}

\subsection{Take-Profit Orders}

\begin{definition}[Take-Profit]
Close position at profit target:
\begin{equation}
\text{Trigger if } s \cdot (P_m - P_e) \geq \text{ProfitTarget}
\end{equation}
\end{definition}

\section{Implementation on Lux}

\subsection{Architecture}

\begin{enumerate}
\item \textbf{Matching Engine}: Order book management and execution
\item \textbf{Position Manager}: Margin, PnL, and liquidation tracking
\item \textbf{Oracle Module}: Index price aggregation
\item \textbf{Funding Engine}: Periodic funding rate calculation and settlement
\item \textbf{Liquidation Engine}: Position monitoring and liquidation execution
\item \textbf{Insurance Fund}: Shortfall coverage
\end{enumerate}

\subsection{Order Matching}

\begin{algorithm}
\caption{Place Limit Order}
\begin{algorithmic}[1]
\Require User $u$, side $s$, price $P$, quantity $q$, leverage $L$
\Ensure Order placed or rejected
\State $M_{\text{required}} \gets q \cdot P / L$
\State \textbf{require} $\text{balance}(u) \geq M_{\text{required}}$
\State \textbf{require} $L \leq L_{\max}(q \cdot P)$
\State Lock margin from user balance
\State Insert order into order book
\State Match against opposing orders if executable
\State Update positions for filled portion
\end{algorithmic}
\end{algorithm}

\subsection{Funding Settlement}

\begin{algorithm}
\caption{Settle Funding (every 8 hours)}
\begin{algorithmic}[1]
\State $F \gets \text{calculateFundingRate}()$
\For{each position $P$}
\State $\text{payment} \gets F \cdot P.\text{notional}$
\If{$P.\text{side} = \text{long}$}
\State $P.\text{margin} \gets P.\text{margin} - \text{payment}$
\Else
\State $P.\text{margin} \gets P.\text{margin} + \text{payment}$
\EndIf
\State Check if position now liquidatable
\EndFor
\end{algorithmic}
\end{algorithm}

\section{Risk Analysis}

\subsection{Oracle Risk}

\begin{definition}[Oracle Deviation Bound]
Maximum acceptable oracle deviation:
\begin{equation}
|\hat{P} - P| < \epsilon \cdot P
\end{equation}
with $\epsilon$ typically 0.1\% to 0.5\%.
\end{definition}

\begin{proposition}[Oracle Manipulation Cost]
For TWAP window $T$ and market depth $D$:
\begin{equation}
\text{Manipulation Cost} \geq \Delta P \cdot D \cdot T
\end{equation}
\end{proposition}

\subsection{Cascading Liquidation Risk}

\begin{definition}[Cascade Risk]
Large liquidations can trigger price moves causing further liquidations:
\begin{equation}
\text{Liquidation Volume} \rightarrow \text{Price Impact} \rightarrow \text{More Liquidations}
\end{equation}
\end{definition}

\begin{proposition}[Cascade Prevention]
Mitigation strategies:
\begin{enumerate}
\item Gradual liquidation (partial close)
\item Liquidation price bands
\item Circuit breakers on extreme moves
\item Insurance fund buffers
\end{enumerate}
\end{proposition}

We have presented a complete framework for perpetual futures on Lux:

\begin{enumerate}
\item \textbf{Funding Rates}: TWAP-based mechanism for price anchoring
\item \textbf{Mark Price}: Manipulation-resistant price for liquidations
\item \textbf{Liquidation Engine}: Efficient position resolution with insurance backstop
\item \textbf{Leverage Management}: Tiered leverage with position limits
\end{enumerate}

Lux's sub-second finality and high throughput make it ideal for perpetual futures trading, enabling responsive liquidations and efficient price discovery.
