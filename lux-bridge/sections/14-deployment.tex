\subsection{Mainnet Contracts}

\textbf{Lux C-Chain (96369)}:
\begin{itemize}[leftmargin=1.1em]
  \item WLUX: \texttt{0x52c84043cd9c865236f11d9fc9f56aa003c1f922}
  \item Bridge Token Base (LRC20B): Deployed per-token with \texttt{MINTER\_ROLE} separation (C-02)
  \item All bridge tokens enforce daily mint limits (C-01) and monotonic nonces (C-03)
\end{itemize}

\textbf{Relayer Network}:
\begin{itemize}[leftmargin=1.1em]
  \item 47 independent relayer operators
  \item Minimum stake: \$100k per relayer
  \item Reward: 0.05\% of bridged volume
  \item Slashing: Full stake loss for fraud
\end{itemize}

\textbf{MPC Committees} (via M-Chain\footnote{See companion paper: ``M-Chain: Decentralized Multi-Party Computation Custody'' \cite{mchain}}):
\begin{itemize}[leftmargin=1.1em]
  \item BTC: 15-node committee (11-of-15 FROST/MuSig2)
  \item ETH: 15-node committee (11-of-15 CGGMP21)
  \item XRPL: 10-node committee (7-of-10 FROST EdDSA)
  \item Minimum stake: 5,000 LUX per MPC signer
\end{itemize}
