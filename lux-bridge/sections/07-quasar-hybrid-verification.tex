\section{Quasar Hybrid Verification}
\label{sec:quasar}
%% +++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++

Quasar is the hybrid consensus verification layer that combines classical BLS12-381 signatures with Pulsar post-quantum lattice signatures, providing security against both classical and quantum adversaries.

\subsection{Dual-Signature Finality Certificates}

\begin{definition}[Quasar Finality Certificate]
A Quasar finality certificate for block $B$ at quantum security phase $\phi \in \{0, 1, 2, 3\}$ is:
\begin{equation}
\mathsf{QFC}_\phi(B) = \begin{cases}
(\sigma_{\text{BLS}}, \text{bitmap}_{\text{BLS}}) & \text{if } \phi = 0 \\
(\sigma_{\text{BLS}}, \sigma_{\text{RT}}, \text{bitmap}_{\text{BLS}}, \text{bitmap}_{\text{RT}}) & \text{if } \phi \in \{1, 2\} \\
(\sigma_{\text{RT}}, \text{bitmap}_{\text{RT}}) & \text{if } \phi = 3
\end{cases}
\end{equation}
where $\sigma_{\text{BLS}}$ is the BLS12-381 aggregate signature and $\sigma_{\text{RT}}$ is the Pulsar lattice threshold signature.
\end{definition}

\begin{theorem}[Quasar Dual-Signature Security]
In Phase 2, a valid Quasar finality certificate requires both a BLS12-381 aggregate signature and a Pulsar threshold signature. An adversary must break both:
\begin{align}
\Pr[\text{Forge}_{\text{Quasar}}] &\leq \Pr[\text{Forge}_{\text{BLS}}] + \Pr[\text{Forge}_{\text{RT}}] \quad \text{(union bound)} \\
&\leq 2^{-128}_{\text{co-CDH}} + 2^{-128}_{\text{LWE}}
\end{align}
Under the stronger independence assumption:
\begin{equation}
\Pr[\text{Forge}_{\text{Quasar}}] \leq \Pr[\text{Forge}_{\text{BLS}}] \cdot \Pr[\text{Forge}_{\text{RT}}] \leq 2^{-256}
\end{equation}
A quantum adversary can break BLS (via Shor on co-CDH) but \textbf{cannot} break Pulsar (LWE is quantum-hard). Therefore, even a quantum adversary faces:
\begin{equation}
\Pr[\text{Forge}_{\text{Quasar, quantum}}] \leq 1 \cdot \Pr[\text{Forge}_{\text{RT}}] \leq 2^{-128}
\end{equation}
\end{theorem}

\subsection{Quasar Verification Algorithm}

\begin{algorithm}[H]
\caption{Quasar Hybrid Finality Verification}
\begin{algorithmic}[1]
\Function{VerifyQuasar}{$\mathsf{QFC}, B, \text{phase}$}
    \State $d \gets \text{Hash}(\text{serialize}(B))$
    \If{$\text{phase} \leq 2$}
        \State \textbf{require} BLS12-381 pairing check: $e(\sigma_{\text{BLS}}, g_2) = e(H(d), \text{apk})$
        \State \textbf{require} signed stake $\geq 2W/3$ from $\text{bitmap}_{\text{BLS}}$
    \EndIf
    \If{$\text{phase} \geq 2$}
        \State \textbf{require} Pulsar verify: $\text{CoronaVerify}(\sigma_{\text{RT}}, d, pk_{\text{RT}})$
        \State \textbf{require} $|\text{bitmap}_{\text{RT}}| \geq t_{\text{RT}}$
    \EndIf
    \State \Return \textsc{valid}
\EndFunction
\end{algorithmic}
\end{algorithm}
