\section{System Model}

\label{sec:wave-model}

\subsection{Network and Adversary}

We model the network as a set of $n$ validators $V = \{v_1, \ldots, v_n\}$ connected
by authenticated point-to-point channels. The network is partially synchronous: there
exists an unknown Global Stabilization Time (GST) after which all messages are delivered
within a known bound $\Delta$.

\begin{definition}[Byzantine Adversary]
The adversary $\mathcal{A}$ controls a set $B \subset V$ of Byzantine validators with
$|B| \leq f < n/3$. Byzantine validators may behave arbitrarily but cannot forge
cryptographic signatures. The adversary is adaptive within an epoch but cannot corrupt
validators instantaneously.
\end{definition}

\begin{definition}[Transaction DAG]
\label{def:wave-dag}
The transaction set $T$ forms a directed acyclic graph $G = (T, E)$ where $(t_i, t_j) \in E$
if $t_j$ is a parent of $t_i$. Each transaction $t$ has a set $\text{parents}(t) \subseteq T$.
The ancestry of $t$ is $\mathcal{A}(t) = \{t' : t' \text{ is reachable from } t \text{ in } G\}$.
\end{definition}

\begin{definition}[Conflict Set]
Transactions $t_i, t_j$ conflict if they consume the same UTXO. The conflict set
$\mathcal{C}(t) = \{t' \in T \setminus \{t\} : t' \text{ conflicts with } t\}$.
Each conflict set induces a binary choice (or $m$-ary for multi-way conflicts) on which
the Wave protocol operates.
\end{definition}

\subsection{Correctness Properties}

\begin{definition}[Safety]
If honest validators $v_i$ and $v_j$ finalize transactions $t$ and $t'$ respectively
where $t \neq t'$, then $t \notin \mathcal{C}(t')$.
\end{definition}

\begin{definition}[Liveness]
If a valid transaction $t$ is submitted and all honest validators eventually receive $t$,
then $t$ or some $t' \notin \mathcal{C}(t)$ is eventually finalized by all honest validators.
\end{definition}
