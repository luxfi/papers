\section{API Reference}


\subsection{Authentication Endpoints}

\begin{lstlisting}[language=bash]
# OAuth 2.0
GET  /v1/iam/authorize
POST /v1/iam/token
POST /v1/iam/revoke
GET  /v1/iam/userinfo

# OpenID Connect
GET  /.well-known/openid-configuration
GET  /.well-known/jwks.json

# SAML
POST /api/saml/acs                # Assertion Consumer Service
GET  /api/saml/metadata           # IdP Metadata
POST /api/saml/slo                # Single Logout

# WebAuthn
GET  /v1/iam/webauthn/signup/begin   # Registration start
POST /v1/iam/webauthn/signup/finish  # Registration complete
GET  /v1/iam/webauthn/signin/begin   # Authentication start
POST /v1/iam/webauthn/signin/finish  # Authentication complete

# Web3 Wallet
POST /v1/iam/web3/challenge          # Get challenge nonce
POST /v1/iam/web3/verify             # Verify signature
\end{lstlisting}
