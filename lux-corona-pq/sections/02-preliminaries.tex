\section{Preliminaries}

\label{sec:prelim}

\subsection{The Cyclotomic Ring}

Let $R = \mathbb{Z}[X]/(X^N + 1)$ and $R_q = \mathbb{Z}_q[X]/(X^N + 1)$ where $N$ is
a power of 2 and $q$ is a prime with $q \equiv 1 \pmod{2N}$ (enabling the number-theoretic
transform, NTT). Elements of $R_q$ are polynomials of degree $< N$ with coefficients in
$\mathbb{Z}_q$; the ring product is polynomial multiplication reduced modulo $X^N + 1$.
For $f \in R_q$ we write $\|f\|_\infty$ for the largest absolute value of its coefficients
(centred representatives in $(-q/2, q/2]$), and $D_\sigma$ for the discrete Gaussian over
$R$ of width $\sigma$ applied coefficient-wise. Corona operates over this \emph{single}
ring $R_q$; all public values, secrets, errors, commitments, challenges and responses are
elements (or short tuples) of $R_q$, never matrices over $R_q$.

\begin{definition}[Ring-LWE]
\label{def:rlwe}
The (decisional) Ring-LWE problem with parameters $(N, q, \sigma)$ is: given a uniformly
random $a \stackrel{\$}{\leftarrow} R_q$ and $b = a \cdot s + e$, where
$s \stackrel{\$}{\leftarrow} D_\sigma$ and $e \stackrel{\$}{\leftarrow} D_\sigma$ are short
and $\cdot$ is ring multiplication in $R_q$, distinguish $(a, b)$ from $(a, u)$ where
$u \stackrel{\$}{\leftarrow} R_q$~\cite{lyubashevsky2010ideal}.
\end{definition}

\begin{remark}[Rank-1 specialisation of Module-LWE]
\label{rem:rank1}
Ring-LWE is exactly Module-LWE specialised to module rank $k = 1$: the public matrix
$\mathbf{A} \in R_q^{M \times N'}$ of the general module construction collapses to a single
ring element $a \in R_q$, and the secret/error vectors collapse to single ring elements
$s, e \in R_q$. Corona is the rank-1 base of the Lux lattice threshold family; Pulsar
(LP-4450) is the rank-$k$ Module-LWE generalisation, lifting $R_q$ to $R_q^k$.
\end{remark}

\begin{definition}[Ring-SIS]
\label{def:rsis}
The Ring-SIS problem with parameters $(N, q, m, \beta)$ is: given uniformly random
$a_1, \ldots, a_m \stackrel{\$}{\leftarrow} R_q$, find $z_1, \ldots, z_m \in R_q$, not all
zero, with $\|z_i\|_\infty \le \beta$ and $\sum_{i=1}^{m} a_i \cdot z_i = 0$ in $R_q$.
Corona's commitments are binding under Ring-SIS, and a forgery yields a short Ring-SIS
solution relative to $a$.
\end{definition}

\subsection{Shamir Secret Sharing over $R_q$}

The secret key is a single short ring element $s \in R_q$. It is shared with one
degree-$(t-1)$ polynomial over the ring.

\begin{definition}[Ring-Based Shamir Sharing]
\label{def:shamir}
Given a secret $s \in R_q$, sample $a_1, \ldots, a_{t-1} \stackrel{\$}{\leftarrow} R_q$ and
form $f(X') = s + \sum_{i=1}^{t-1} a_i (X')^i \in R_q[X']$, evaluated at distinct public
points $X' = \mathrm{id}_i \in \mathbb{Z}_q$ (the indeterminate $X'$ of the sharing
polynomial is distinct from the ring indeterminate $X$). Party $i$ receives the share
$s_i = f(\mathrm{id}_i) \in R_q$.
\end{definition}

Lagrange interpolation over $R_q$ reconstructs $s$ from any $t$ shares:
\begin{equation}
s = \sum_{i \in S} \lambda_i \cdot s_i, \qquad
\lambda_i = \prod_{j \in S,\, j \neq i} \frac{\mathrm{id}_j}{\mathrm{id}_j - \mathrm{id}_i} \in \mathbb{Z}_q
\label{eq:lagrange}
\end{equation}
where the Lagrange coefficients $\lambda_i$ are scalars in $\mathbb{Z}_q$ (embedded in $R_q$).
