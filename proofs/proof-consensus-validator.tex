\begin{abstract}
We prove properties of the validator staking and slashing model: slashing is loss-bounded (cannot exceed stake), slashing is monotone, honest validators are never slashed, and higher stake yields more voting power.
\end{abstract}

\section{Introduction}
This module formalizes the economic incentive layer of Lux consensus. Validators stake tokens and receive voting power proportional to their stake. Provable misbehavior results in slashing, but the slashed amount is always bounded by the validator's stake. The model maps to \texttt{lux/node/vms/platformvm} (validator staking) and the validator economics white paper.

\section{Definitions}
\begin{definition}[Validator]
A validator with id, stake, honesty flag, and slashed amount.
\end{definition}

\begin{definition}[ValidatorSet]
Set of validators with minimum stake and total stake.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{slash\_bounded}]
Slashing penalty cannot cause total slashed amount to exceed the validator's stake: $(\text{slash}~v~p).\text{slashed} \leq v.\text{stake}$.
\end{theorem}
\begin{proof}
By \texttt{simp} and \texttt{omega} on the \texttt{min} in the slash function.
\end{proof}

\begin{theorem}[\texttt{slash\_monotone}]
Slashed amount only increases: $(\text{slash}~v~p).\text{slashed} \geq v.\text{slashed}$.
\end{theorem}
\begin{proof}
By \texttt{omega}.
\end{proof}

\begin{axiom_stmt}[\texttt{honest\_no\_slash}]
Honest validators always have $\text{slashed} = 0$. This is a protocol-level guarantee.
\end{axiom_stmt}

\begin{theorem}[\texttt{below\_minimum\_rejected}]
A validator with stake below the minimum fails the \texttt{meetsMinimum} check.
\end{theorem}

\begin{theorem}[\texttt{more\_stake\_more\_power}]
If $v_1.\text{stake} > v_2.\text{stake}$, then $\text{votingPower}(v_1) \geq \text{votingPower}(v_2)$.
\end{theorem}
\begin{proof}
By monotonicity of integer division.
\end{proof}

\begin{theorem}[\texttt{empty\_zero\_stake}]
$\text{computeTotalStake}([]) = 0$.
\end{theorem}
