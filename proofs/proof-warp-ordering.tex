\title{Formal Verification: Warp Causal Message Ordering}
\author{Zach Kelling, Lux Industries}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We prove that Warp messages respect causal ordering: if message $A$ precedes message $B$ on the source chain (by block height or transaction index), then $A$ is processed before $B$ on the destination chain. The ordering relation is transitive and irreflexive.
\end{abstract}

\section{Introduction}
Causal ordering ensures that cross-chain state updates are applied in the correct order. This is critical for the treasury system (\texttt{Collect.sol}) where fee governance settings must be applied monotonically.

\section{Definitions}
\begin{definition}[OrderedMessage]
A message with block height, transaction index, nonce, and payload.
\end{definition}

\begin{definition}[VersionedState]
Version-based state for monotonic updates (treasury settings).
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{version\_monotone}]
Accepted versions strictly increase: $s'.\text{version} > s.\text{version}$.
\end{theorem}

\begin{theorem}[\texttt{stale\_rejected}]
Stale versions ($v \leq s.\text{version}$) are rejected.
\end{theorem}

\begin{theorem}[\texttt{precedes\_trans}]
The precedence relation is transitive.
\end{theorem}
\begin{proof}
By case analysis on block height and transaction index comparisons.
\end{proof}

\begin{theorem}[\texttt{precedes\_irrefl}]
The precedence relation is irreflexive: no message precedes itself.
\end{theorem}
