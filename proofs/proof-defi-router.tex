\title{Formal Verification: Cross-Pool Order Router}
\author{Zach Kelling \and Woo Bin}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We formalize the order routing protocol that splits trades across multiple AMM pools for best execution. Splitting reduces slippage compared to a single-pool trade.
\end{abstract}

\section{Introduction}
The router aggregates liquidity across multiple pools to achieve best execution. For large orders, splitting across pools reduces price impact compared to executing against a single pool.

\section{Definitions}
\begin{definition}[Pool]
A liquidity pool with reserves and fee.
\end{definition}

\begin{definition}[Route]
A route: list of (pool, amount) pairs with total input.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{split\_better}]
Splitting reduces slippage: full output $\leq$ reserve.
\end{theorem}
\begin{proof}
By division bound.
\end{proof}

\begin{theorem}[\texttt{empty\_route\_zero}]
Empty route yields zero output.
\end{theorem}

\begin{theorem}[\texttt{route\_atomic}]
Route execution is atomic (single transaction).
\end{theorem}

\begin{theorem}[\texttt{more\_pools\_help}]
Adding pools increases routing options.
\end{theorem}
