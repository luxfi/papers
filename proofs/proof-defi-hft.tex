\author{Zach Kelling, Lux Industries}

\begin{abstract}
We formalize the geographic moat of the Lux HFT venue in Kansas City. On-premise validators trade at $\sim$20$\mu$s RTT, capturing 100\% of arbitrage profits on tight spreads. Remote participants (NYC, London, Beijing) face latency costs exceeding gross profits.
\end{abstract}

\section{Introduction}
The HFT venue model proves that geographic proximity creates a structural advantage. Only on-premise validators at the Kansas City venue can profitably capture tight-spread arbitrage. This is the economic moat that incentivizes validator co-location and drives consensus participation.

\section{Definitions}
\begin{definition}[VenueParams]
Venue network: 10 Gbps external, 200 Gbps internal, 20$\mu$s local RTT.
\end{definition}

\begin{definition}[ArbOpp]
Arbitrage opportunity: spread, volume, decay rate per millisecond.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{local\_zero\_cost}]
On-premise latency cost is zero: 20$\mu$s $< 1$ms.
\end{theorem}

\begin{theorem}[\texttt{local\_full\_capture}]
On-premise net profit equals gross profit.
\end{theorem}

\begin{theorem}[\texttt{closer\_more\_profit}]
Closer location yields more profit: $\text{netProfit}(r_1) \geq \text{netProfit}(r_2)$ when $r_1 < r_2$.
\end{theorem}

\begin{theorem}[\texttt{kc\_daily}]
KC venue daily gross on typical arb: \$5,000.
\end{theorem}

\begin{theorem}[\texttt{nyc\_negative}]
NYC (18ms RTT) is net negative on typical arb.
\end{theorem}

\begin{theorem}[\texttt{extended\_profit\_ordering}]
Profit ordering: KC $>$ Chicago $>$ NYC $>$ London $>$ Beijing $>$ Singapore.
\end{theorem}

\begin{theorem}[\texttt{colocation\_beats\_remote}]
Co-located validators beat any remote participant with RTT $> 1$ms.
\end{theorem}
