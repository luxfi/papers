\title{Formal Verification: Peer Discovery and Eclipse Resistance}
\author{Zach Kelling \and Woo Bin}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We formalize the peer discovery protocol with bounded degree, eclipse attack resistance via bootstrap nodes, and gossip convergence. The peer set has a maximum size to prevent resource exhaustion.
\end{abstract}

\section{Introduction}
The peer discovery model ensures bounded resource usage and resistance to eclipse attacks. Bootstrap connections provide minimum connectivity, and the bounded degree prevents a single malicious node from consuming all connection slots.

\section{Definitions}
\begin{definition}[PeerSet]
Peer set state: current peers, max peers, bootstrap connections.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{bounded\_degree}]
Peer count never exceeds max: $(\text{addPeer}~s).\text{peers} \leq s.\text{maxPeers}$.
\end{theorem}
\begin{proof}
By case split on capacity check.
\end{proof}

\begin{theorem}[\texttt{add\_monotone}]
Peer count only increases on add.
\end{theorem}

\begin{theorem}[\texttt{remove\_monotone}]
Peer count only decreases on remove.
\end{theorem}

\begin{theorem}[\texttt{at\_capacity\_no\_add}]
Full peer set rejects new connections: peers remain unchanged when at capacity.
\end{theorem}
