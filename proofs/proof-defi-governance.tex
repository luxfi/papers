\title{Formal Verification: On-Chain DEX Governance}
\author{Zach Kelling \and Woo Bin}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We formalize the DAO governance model for DEX parameter changes. Proposals require both quorum (minimum votes) and timelock (mandatory delay) before execution. Fee changes are bounded to prevent abuse.
\end{abstract}

\section{Introduction}
The governance model ensures that DEX parameter changes are subject to democratic approval with safety bounds. The timelock prevents flash-governance attacks, and fee bounds prevent parameter manipulation.

\section{Definitions}
\begin{definition}[ParamChange]
Types: fee update, pool weight, listing, delisting, upgrade.
\end{definition}

\begin{definition}[Proposal]
A proposal with votes, quorum, timelock, and timestamp.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{below\_quorum\_fails}]
Below-quorum proposals are rejected.
\end{theorem}

\begin{theorem}[\texttt{early\_execution\_blocked}]
Before timelock expires, execution is blocked.
\end{theorem}

\begin{theorem}[\texttt{both\_required}]
Execution requires both passing AND timelock satisfaction.
\end{theorem}
\begin{proof}
Decompose the Boolean conjunction.
\end{proof}

\begin{theorem}[\texttt{fee\_bounded}]
Fee changes exceeding 1\% (100 bps) are rejected.
\end{theorem}

\begin{theorem}[\texttt{majority\_required}]
Passing requires more votes for than against.
\end{theorem}
