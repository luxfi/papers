\title{Formal Verification: Flash Loan Safety}
\author{Zach Kelling \and Woo Bin}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We prove that flash loans preserve pool solvency: the pool balance after a flash loan is always $\geq$ the balance before. If repayment fails, the entire transaction reverts with no state change.
\end{abstract}

\section{Introduction}
Flash loans enable atomic borrow-use-repay within a single transaction. The solvency invariant ensures that pools are protected: either the loan is fully repaid with fee, or the entire transaction reverts.

\section{Definitions}
\begin{definition}[LoanState]
Pool state: balance and fee in basis points.
\end{definition}

\begin{definition}[FlashLoan]
A flash loan: amount, fee, repaid flag.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{flash\_loan\_solvent}]
Pool never loses money: $(\text{executeLoan}~s~\text{amt}~r).\text{balance} \geq s.\text{balance}$.
\end{theorem}
\begin{proof}
By case split on repaid flag.
\end{proof}

\begin{theorem}[\texttt{no\_repay\_reverts}]
Unreturned loan reverts: $\text{executeLoan}(s, \text{amt}, \text{false}) = s$.
\end{theorem}

\begin{theorem}[\texttt{fee\_accrues}]
Successful loan increases pool by fee.
\end{theorem}

\begin{theorem}[\texttt{overdraft\_blocked}]
Cannot borrow more than pool has: loan exceeding balance reverts.
\end{theorem}
