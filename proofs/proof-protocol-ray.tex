\author{Zach Kelling \and Woo Bin}
\date{December 2025}

\begin{abstract}
We formalize Ray, the linear chain consensus driver that processes items sequentially from a source and commits decided items to a sink in order. We prove that decisions advance a prefix frontier with no gaps and the decided count is monotonically non-decreasing.
\end{abstract}

\section{Introduction}
Ray is the linear consensus driver that wraps Wave for sequential block processing. It processes items from a Source, runs Wave voting on each, and commits to a Sink in strict order. The no-gaps invariant ensures that decided items form a contiguous prefix of the source order.

\section{Definitions}
\begin{definition}[RayState]
State tracking source order, decided count (prefix length), and per-item decisions.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{decideNext\_marks\_item}]
The \texttt{decideNext} function marks exactly the next item as decided.
\end{theorem}

\begin{theorem}[\texttt{decideNext\_advances}]
\texttt{decideNext} advances the frontier by exactly 1 when items remain.
\end{theorem}

\begin{theorem}[\texttt{init\_nothing\_decided}]
Initial state has decided count 0.
\end{theorem}

\begin{theorem}[\texttt{no\_gaps}]
After a step, the decided count is either $s.\text{decided}+1$ or $s.\text{decided}$ (no gaps).
\end{theorem}

\begin{theorem}[\texttt{decided\_monotone}]
The decided count never decreases: $(\text{decideNext}~s~d).\text{decided} \geq s.\text{decided}$.
\end{theorem}
\begin{proof}
By \texttt{omega} on both branches of the match.
\end{proof}
