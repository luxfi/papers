\title{Formal Verification: Build Reproducibility}
\author{Zach Kelling \and Woo Bin}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We prove the main reproducibility theorem: a build is reproducible if and only if it avoids non-deterministic accesses (time, random, GPU, non-content-addressed I/O). Reproducibility composes: two reproducible builds composed sequentially yield a reproducible build. One non-reproducible access contaminates the entire build.
\end{abstract}

\section{Introduction}
The reproducibility theorem connects the coeffect algebra to practical build correctness. It provides a machine-checkable criterion for determining whether a build will produce identical outputs across different environments, which is essential for supply chain security and content-addressed distribution.

\section{Definitions}
\begin{definition}[Access]
Access types: pure, CA file, CA network, non-CA file, non-CA network, time, random, GPU.
\end{definition}

\begin{definition}[BuildTrace]
A build trace: sequence of accesses.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{empty\_reproducible}]
Empty build is reproducible.
\end{theorem}

\begin{theorem}[\texttt{time\_breaks}]
Time access breaks reproducibility.
\end{theorem}

\begin{theorem}[\texttt{random\_breaks}]
Random access breaks reproducibility.
\end{theorem}

\begin{theorem}[\texttt{repro\_compose}]
Reproducible builds compose: if $a$ and $b$ are reproducible, then $a \mathbin{++} b$ is reproducible.
\end{theorem}
\begin{proof}
By \texttt{List.all\_append} and Boolean conjunction.
\end{proof}

\begin{theorem}[\texttt{one\_bad\_contaminates}]
One non-reproducible access contaminates the entire build: $\text{isReproducible}(\text{prefix} \mathbin{++} [\text{time}] \mathbin{++} \text{suffix}) = \text{false}$.
\end{theorem}

\begin{axiom_stmt}[\texttt{repro\_deterministic}]
Reproducible builds with the same inputs produce the same outputs.
\end{axiom_stmt}

\begin{theorem}[\texttt{repro\_zero\_bad}]
A reproducible trace has zero non-reproducible accesses.
\end{theorem}
\begin{proof}
By \texttt{List.filter\_eq\_nil} and the \texttt{all} predicate.
\end{proof}
