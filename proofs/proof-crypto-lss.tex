\begin{abstract}
We formalize Linear Secret Sharing (LSS), the foundation for all threshold cryptography in Lux. LSS generalizes Shamir's secret sharing to linear access structures. We state correctness ($t$ shares reconstruct), privacy ($t-1$ shares reveal nothing), and linearity (local addition without interaction).
\end{abstract}

\section{Introduction}
LSS is the algebraic foundation upon which FROST, CMP (CGGMP21), TFHE threshold decryption, and Ringtail threshold signatures are all built. The linearity property enables MPC addition without interaction --- each party simply adds their shares locally. This is why LSS is the base layer for all Lux threshold protocols.

\section{Definitions}
\begin{definition}[AccessStructure]
Access structure: $n$ total parties, threshold $t$, with $t \leq n$ and $0 < t$.
\end{definition}

\begin{definition}[Share]
A share with index and value.
\end{definition}


\section{Theorems and Axioms}
\begin{axiom_stmt}[\texttt{lss\_correctness}]
Any $t$ shares reconstruct the original secret.
\end{axiom_stmt}

\begin{axiom_stmt}[\texttt{lss\_privacy}]
$t-1$ shares reveal nothing about the secret (information-theoretic security).
\end{axiom_stmt}

\begin{axiom_stmt}[\texttt{lss\_linearity}]
$\text{share}(a+b) = \text{share}(a) + \text{share}(b)$ (computed locally per party).
\end{axiom_stmt}

\begin{theorem}[\texttt{shamir\_is\_lss}]
Shamir's scheme (degree-$(t-1)$ polynomial evaluation) is a canonical LSS instance.
\end{theorem}

\begin{theorem}[\texttt{protocol\_composability}]
Having $k \geq t$ shares suffices for any threshold protocol built on this access structure.
\end{theorem}

\begin{axiom_stmt}[\texttt{reshare\_correctness}]
Resharing creates new shares of the same secret without revealing it.
\end{axiom_stmt}

\begin{axiom_stmt}[\texttt{vss\_honest\_valid}]
Verifiable secret sharing: honestly generated shares pass verification.
\end{axiom_stmt}

\begin{theorem}[\texttt{proactive\_refresh\_resets}]
Proactive refresh requires $t$ shares within each epoch, not across epochs.
\end{theorem}
