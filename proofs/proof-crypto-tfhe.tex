\begin{abstract}
We formalize TFHE (Torus FHE), the core of Lux's confidential compute layer. TFHE supports Boolean gates (AND, OR, NOT, NAND) on encrypted data with bounded noise growth. Bootstrapping resets noise, enabling unlimited computation depth. Programmable bootstrapping evaluates arbitrary functions during the bootstrap step itself.
\end{abstract}

\section{Introduction}
TFHE enables Turing-complete computation on encrypted data. The NAND gate universality (any Boolean circuit from NAND gates) combined with programmable bootstrapping makes TFHE the foundation for confidential smart contracts, private voting, and encrypted ML inference on Lux.

\section{Definitions}
\begin{definition}[Params]
TFHE parameters: LWE dimension $n$, RLWE dimension $N$, modulus $q$, noise $\sigma$.
\end{definition}

\begin{definition}[Ciphertext]
An encrypted value with current noise level and validity flag.
\end{definition}


\section{Theorems and Axioms}
\begin{axiom_stmt}[\texttt{tfhe\_correctness}]
Decrypt of encrypt equals the original plaintext when noise is valid.
\end{axiom_stmt}

\begin{axiom_stmt}[\texttt{hom\_and\_correct}]
Homomorphic AND: $\text{decrypt}(\text{AND}(\text{enc}(a), \text{enc}(b))) = a \wedge b$.
\end{axiom_stmt}

\begin{axiom_stmt}[\texttt{hom\_nand\_correct}]
Homomorphic NAND: $\text{decrypt}(\text{NAND}(\text{enc}(a), \text{enc}(b))) = \neg(a \wedge b)$.
\end{axiom_stmt}

\begin{axiom_stmt}[\texttt{and\_noise\_growth}]
AND noise bound: $(\text{AND}~c_a~c_b).\text{noise} \leq c_a.\text{noise} + c_b.\text{noise} + 1$.
\end{axiom_stmt}

\begin{axiom_stmt}[\texttt{not\_noise\_free}]
NOT is noiseless: $(\text{NOT}~c).\text{noise} = c.\text{noise}$.
\end{axiom_stmt}

\begin{axiom_stmt}[\texttt{bootstrap\_preserves}]
Bootstrapping preserves the encrypted value.
\end{axiom_stmt}

\begin{theorem}[\texttt{bootstrap\_restores\_budget}]
Bootstrapping restores the noise budget to at least the fresh encryption level.
\end{theorem}
\begin{proof}
By \texttt{omega} on the noise reset axiom.
\end{proof}

\begin{axiom_stmt}[\texttt{programmable\_bootstrap\_correct}]
Programmable bootstrapping: $\text{decrypt}(\text{PBS}(c, f)) = f(\text{decrypt}(c))$.
\end{axiom_stmt}

\begin{theorem}[\texttt{threshold\_privacy}]
Fewer than $t$ decryption shares reveal nothing about the plaintext.
\end{theorem}
