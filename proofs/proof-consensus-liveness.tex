\title{Formal Verification: Consensus Liveness}
\author{Zach Kelling \and Woo Bin}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We prove that under partial synchrony with $f < n/3$ Byzantine nodes, the Lux consensus protocol eventually finalizes a value within $\beta + \delta$ rounds after the Global Stabilization Time (GST). The proof uses a partial synchrony network model and shows that sustained honest majority preference leads to confidence accumulation and eventual finalization.
\end{abstract}

\section{Introduction}
The liveness proof complements the safety theorem by showing that consensus makes progress. After GST, honest nodes can communicate reliably, enabling sustained confidence accumulation. The expected time to finality is $O(\beta)$ rounds, matching the implementation in \texttt{protocol/wave/wave.go} and \texttt{protocol/ray/ray.go}.

\section{Definitions}
\begin{definition}[NetworkModel]
Partial synchrony model with GST (Global Stabilization Time) and $\delta$ (max message delay after GST).
\end{definition}

\begin{definition}[StateTrace]
Sequence of consensus states indexed by round number.
\end{definition}


\section{Theorems and Axioms}
\begin{axiom_stmt}[\texttt{post\_gst\_delivery}]
After GST, all messages between honest nodes arrive within $\delta$ rounds.
\end{axiom_stmt}

\begin{axiom_stmt}[\texttt{honest\_sample\_dominance}]
After GST, if honest nodes have $\alpha$-majority for value $v$, that majority persists in the next round.
\end{axiom_stmt}

\begin{axiom_stmt}[\texttt{confidence\_accumulation}]
Under synchrony with $\alpha$-majority, each honest node's confidence increments. Maps to \texttt{wave.go:159: state.Count++}.
\end{axiom_stmt}

\begin{axiom_stmt}[\texttt{beta\_triggers\_finalization}]
When confidence reaches $\beta$, finalization occurs. Maps to \texttt{wave.go:161--163}.
\end{axiom_stmt}

\begin{theorem}[\texttt{liveness}]
After GST, if honest count $> 2f$ and a common preference has $\alpha$-majority sustained for $\beta$ rounds, then there exists a round $r \leq r_0 + \beta + \delta$ and an honest node $i$ that has finalized value $v$.
\end{theorem}
\begin{proof}
Obtain an honest node with a preference from \texttt{honest\_node\_exists}. Apply \texttt{sustained\_confidence} over $\beta$ synchronous rounds to reach confidence $\geq \beta$. Apply \texttt{beta\_triggers\_finalization} to conclude finalization.
\end{proof}
