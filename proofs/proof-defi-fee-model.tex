\title{Formal Verification: DeFi Fee Model}
\author{Zach Kelling \and Woo Bin}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We formalize the maker-taker fee model with volume-based tiers. The key properties are: maker fees are always $\leq$ taker fees (incentivizing liquidity provision), higher volume yields lower fees, and validator revenue is bounded.
\end{abstract}

\section{Introduction}
The fee model incentivizes liquidity provision (lower maker fees) and rewards high-volume participants (volume tiers). The permissionless fee schedule is on-chain and transparent.

\section{Definitions}
\begin{definition}[FeeTier]
A fee tier: minimum volume, maker basis points, taker basis points.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{maker\_le\_taker}]
Maker fee $\leq$ taker fee for any tier where $\text{makerBps} \leq \text{takerBps}$.
\end{theorem}
\begin{proof}
By monotonicity of integer division.
\end{proof}

\begin{theorem}[\texttt{top\_tier\_zero}]
Top tier ($\$100$M+) has zero maker fee.
\end{theorem}

\begin{theorem}[\texttt{volume\_discount}]
Higher volume tiers have lower taker fees.
\end{theorem}

\begin{theorem}[\texttt{four\_tiers}]
The fee schedule has exactly 4 tiers.
\end{theorem}

\begin{theorem}[\texttt{validator\_bounded}]
Validator share of fees never exceeds total fees.
\end{theorem}
\begin{proof}
By division bound.
\end{proof}
