\author{Zach Kelling, Lux Industries}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We formalize the on-chain order book with price-time priority, no self-trading, and deterministic matching. Fill quantities are bounded by order sizes and the matching engine is deterministic.
\end{abstract}

\section{Introduction}
The order book model ensures fair, deterministic execution for the Lux DEX. Price-time priority rewards early liquidity providers, and the no-self-trade rule prevents wash trading.

\section{Definitions}
\begin{definition}[Order]
An order: id, trader, side (bid/ask), price, quantity, timestamp.
\end{definition}

\begin{definition}[Fill]
A matched trade: buy order, sell order, price, quantity.
\end{definition}

\begin{definition}[BookState]
Order book: sorted bids, sorted asks, fills.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{empty\_no\_match}]
Empty book produces no fills.
\end{theorem}

\begin{theorem}[\texttt{deterministic}]
Same state yields same result: $\text{tryMatch}(b) = \text{tryMatch}(b)$.
\end{theorem}

\begin{theorem}[\texttt{buyer\_gets\_price\_or\_better}]
Fill price $\leq$ bid price (buyer gets price improvement).
\end{theorem}

\begin{theorem}[\texttt{fill\_bounded}]
Fill quantity $\leq \min(\text{bid\_qty}, \text{ask\_qty})$.
\end{theorem}
