\begin{abstract}
We formalize Field, the DAG-mode consensus driver. Field uses Wave for voting and Flare for commit/skip classification. The key property is that committed vertices form a consistent cut of the DAG: if vertex $v$ is committed and $u$ is an ancestor of $v$, then $u$ is also committed.
\end{abstract}

\section{Introduction}
Field is the DAG-mode counterpart to Ray. While Ray processes blocks linearly, Field operates on a DAG and uses the consistent cut property to determine safe commit prefixes. This maps to \texttt{protocol/field/field.go} and \texttt{core/dag/horizon.go} (\texttt{ComputeSafePrefix}).

\section{Definitions}
\begin{definition}[DAG]
A directed acyclic graph with vertices, parent relation, and a closure property.
\end{definition}

\begin{definition}[IsAncestor]
Inductive ancestry relation: reflexive and transitive via parent edges.
\end{definition}

\begin{definition}[IsConsistentCut]
A consistent cut is a downward-closed subset under ancestry.
\end{definition}

\begin{definition}[FieldState]
State: DAG, committed set, frontier set.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{committed\_is\_consistent\_cut}]
After committing a prefix, the committed set remains a consistent cut.
\end{theorem}
\begin{proof}
By construction: the \texttt{commitPrefix} function requires a proof that the union is a consistent cut.
\end{proof}

\begin{theorem}[\texttt{committed\_monotone}]
The committed set is monotonically growing: $s.\text{committed} \subseteq (\text{commitPrefix}~s~\text{nc}).\text{committed}$.
\end{theorem}
\begin{proof}
By \texttt{Finset.subset\_union\_left}.
\end{proof}

\begin{theorem}[\texttt{empty\_consistent}]
The empty DAG has a trivially consistent empty cut.
\end{theorem}

\begin{theorem}[\texttt{new\_commits\_included}]
New commits are included in the new committed set.
\end{theorem}
\begin{proof}
By \texttt{Finset.subset\_union\_right}.
\end{proof}
