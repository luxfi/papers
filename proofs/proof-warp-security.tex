\title{Formal Verification: Warp Message Security}
\author{Zach Kelling \and Woo Bin}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We prove the security properties of Warp cross-chain messages: authentication via BLS quorum, replay protection via nonces, and quorum intersection ensuring that conflicting messages cannot both be authenticated.
\end{abstract}

\section{Introduction}
The Warp security model ensures that cross-chain messages are trustworthy. BLS aggregate signatures provide authentication, nonces prevent replay, and quorum intersection prevents conflicting messages from passing verification.

\section{Definitions}
\begin{definition}[WarpMessage]
A Warp message with source/dest chains, nonce, payload, signer count, and quorum.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{auth\_requires\_quorum}]
Authentication requires signer count $\geq$ quorum.
\end{theorem}

\begin{theorem}[\texttt{insufficient\_sigs}]
Below-quorum messages are not authenticated.
\end{theorem}

\begin{theorem}[\texttt{replay\_protection}]
Messages with different nonces from the same source are distinct.
\end{theorem}

\begin{theorem}[\texttt{warp\_quorum\_overlap}]
Two authenticated messages from the same source chain share at least one honest signer: $\text{signers}_1 + \text{signers}_2 > n$.
\end{theorem}
\begin{proof}
By quorum arithmetic with $3f < n$ and quorum $= 2f+1$.
\end{proof}
