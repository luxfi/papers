\title{Formal Verification: Post-Quantum Key Exchange Handshake}
\author{Zach Kelling, Lux Industries}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We formalize the hybrid key exchange protocol combining X25519 (classical) and ML-KEM-768 (post-quantum). The session key is derived from both shared secrets, ensuring that an attacker must break both to compromise a session.
\end{abstract}

\section{Introduction}
The hybrid handshake provides quantum-safe key exchange for Lux node-to-node communication. Even if X25519 is broken by a quantum computer, ML-KEM-768 (FIPS 203) protects the session. The transcript binding ensures the session key commits to the full handshake history.

\section{Definitions}
\begin{definition}[State]
Handshake state machine: initial, sentHello, receivedHello, established, failed.
\end{definition}

\begin{definition}[KeyExchange]
Key exchange components: X25519 shared secret, ML-KEM shared secret, transcript hash.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{hybrid\_both\_contribute}]
The session key depends on both shared secrets: $\text{deriveSessionKey}(\text{kex}) > \text{kex.x25519SS}$ when both components are positive.
\end{theorem}
\begin{proof}
By \texttt{omega}.
\end{proof}

\begin{axiom_stmt}[\texttt{forward\_secrecy}]
Each session uses fresh ephemeral keys. Different transcript hashes yield different session keys.
\end{axiom_stmt}

\begin{theorem}[\texttt{initial\_no\_key}]
The initial state has no session key.
\end{theorem}

\begin{theorem}[\texttt{established\_has\_key}]
The established state contains the session key.
\end{theorem}
