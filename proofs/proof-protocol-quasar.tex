\title{Formal Verification: Quasar Protocol: BLS + Ringtail Hybrid Finality}
\author{Zach Kelling, Lux Industries}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We formalize the Quasar protocol, the quantum finality engine that produces dual-signed (BLS + Ringtail) aggregate blocks. We prove BLS aggregation soundness, quorum intersection, dual signature requirements, and epoch monotonicity.
\end{abstract}

\section{Introduction}
Quasar operates at the protocol level, collecting blocks from all Lux chains (P, C, X, A, etc.) and producing quantum-final aggregate blocks. Each block carries both BLS (fast, classical) and Ringtail (post-quantum, lattice-based) signatures, ensuring quantum resistance.

\section{Definitions}
\begin{definition}[BLSSig / RingtailSig]
Abstract signature types with validity flags.
\end{definition}

\begin{definition}[QuasarSig]
Dual signature: BLS + Ringtail components.
\end{definition}

\begin{definition}[QuantumBlock]
A quantum block with height, source block hashes, and dual signature.
\end{definition}

\begin{definition}[QuasarState]
State: quantum height and list of finalized quantum blocks.
\end{definition}


\section{Theorems and Axioms}
\begin{axiom_stmt}[\texttt{bls\_aggregation\_sound}]
Aggregation of valid BLS signatures produces a valid aggregate (bilinear pairing property).
\end{axiom_stmt}

\begin{theorem}[\texttt{height\_monotone}]
Quantum height is strictly increasing when a new block is added.
\end{theorem}
\begin{proof}
By \texttt{simp} on \texttt{addBlock}.
\end{proof}

\begin{theorem}[\texttt{bls\_quorum\_intersection}]
Two quorums of $\geq 2f+1$ from $n$ validators with $3f < n$ overlap: $q_1 + q_2 > n$.
\end{theorem}
\begin{proof}
By \texttt{omega}.
\end{proof}

\begin{theorem}[\texttt{quantum\_finality\_requires\_both}]
A valid quantum signature requires both BLS and Ringtail components to be valid.
\end{theorem}
\begin{proof}
Decompose the Boolean conjunction.
\end{proof}
