\title{Formal Verification: Wave Threshold Voting Engine}
\author{Zach Kelling \and Woo Bin}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We formalize the Wave protocol, the core threshold voting engine of Lux consensus. Each round samples $k$ peers, compares votes against an $\alpha$ threshold, and accumulates confidence toward the $\beta$ finalization target. We prove that decided items are stable, confidence is monotone on matching rounds, and finalization occurs after $\beta$ consecutive confirming rounds.
\end{abstract}

\section{Introduction}
Wave is the threshold voting engine at the core of Lux consensus. It implements the sampling-based voting loop described in \texttt{protocol/wave/wave.go}. The FPC variant selects $\alpha = \lceil\theta \cdot k\rceil$ where $\theta$ is PRF-selected per phase, providing additional Byzantine resilience.

\section{Definitions}
\begin{definition}[WaveConfig]
Configuration: $k$ (sample size), $\alpha$ (threshold), $\beta$ (confidence target), with $0 < k$, $\alpha \leq k$, $0 < \beta$.
\end{definition}

\begin{definition}[WaveState]
Per-item state: decided flag, preference (commit/skip), confidence counter.
\end{definition}

\begin{definition}[RoundResult]
A round outcome: yes votes and total votes.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{decided\_is\_stable}]
Once decided, the step function returns the same state: $\text{step}(\text{cfg}, s, r) = s$ when $s.\text{decided} = \text{true}$.
\end{theorem}
\begin{proof}
By \texttt{simp} on the early-return branch.
\end{proof}

\begin{theorem}[\texttt{confidence\_monotone\_on\_match}]
When preference matches threshold ($\text{yesVotes} \geq \alpha$ and preference is true), confidence increments by 1.
\end{theorem}
\begin{proof}
By case analysis on the step function branches.
\end{proof}

\begin{theorem}[\texttt{finalization\_after\_beta}]
When confidence $+1 \geq \beta$ and the threshold is met with matching preference, the item is decided.
\end{theorem}
\begin{proof}
By \texttt{simp} on the step function and \texttt{omega}.
\end{proof}
