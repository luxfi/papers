\title{Formal Verification: ML-KEM (FIPS 203) Key Encapsulation}
\author{Zach Kelling, Lux Industries}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We formalize ML-KEM, the post-quantum key encapsulation mechanism based on CRYSTALS-Kyber. ML-KEM-768 is the default for Lux hybrid key exchange. We state correctness (decapsulation recovers the shared secret) and IND-CCA2 security as axioms.
\end{abstract}

\section{Introduction}
ML-KEM is used in the Lux hybrid handshake protocol (see \texttt{Protocol/Handshake.lean}). Combined with X25519, it provides a belt-and-suspenders approach: breaking the session key requires breaking both classical ECDH and post-quantum lattice problems.

\section{Definitions}
\begin{definition}[SecurityLevel]
Three levels: ML-KEM-512, ML-KEM-768 (default), ML-KEM-1024.
\end{definition}


\section{Theorems and Axioms}
\begin{axiom_stmt}[\texttt{mlkem\_correctness}]
Decapsulation of a valid encapsulation recovers the original shared secret: $\text{decaps}(\text{sk}, \text{encaps}(\text{pk}).\text{ct}) = \text{encaps}(\text{pk}).\text{ss}$.
\end{axiom_stmt}

\begin{axiom_stmt}[\texttt{mlkem\_ind\_cca2}]
IND-CCA2 security: ciphertexts are indistinguishable under Module-LWE hardness.
\end{axiom_stmt}

\begin{theorem}[\texttt{ss\_fixed\_size}]
Shared secret is always 32 bytes regardless of security level.
\end{theorem}

\begin{theorem}[\texttt{pk\_monotone\_512\_768}]
Public key sizes grow with security level: $\text{pkSize}(512) < \text{pkSize}(768)$.
\end{theorem}

\begin{theorem}[\texttt{ct\_bounded}]
Ciphertext size is bounded: $\leq 1568$ bytes for all levels.
\end{theorem}

\begin{axiom_stmt}[\texttt{hybrid\_kem\_combines\_both}]
The hybrid shared secret is derived from both ML-KEM and X25519 components.
\end{axiom_stmt}
