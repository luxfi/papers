\title{Formal Verification: Flare DAG Commit Rule}
\author{Zach Kelling \and Woo Bin}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We formalize the Flare protocol, the DAG commit rule for Lux. For a proposer vertex at round $r$, Flare classifies it as Commit ($\geq 2f+1$ support), Skip ($\geq 2f+1$ non-support), or Undecided. We prove that Commit and Skip are mutually exclusive, certificates require honest support, and classification is total.
\end{abstract}

\section{Introduction}
Flare is the DAG commit rule used in the Nebula (DAG mode) consensus path. It maps to \texttt{core/dag/flare.go} where \texttt{HasCertificate} requires $\geq 2f+1$ support and \texttt{HasSkip} requires $\geq 2f+1$ non-support. The mutual exclusivity proof is critical for DAG safety.

\section{Definitions}
\begin{definition}[Params]
DAG parameters: $n$ nodes, $f$ max Byzantine, with $3f < n$.
\end{definition}

\begin{definition}[VoteCounts]
Support and non-support counts at round $r+1$ for a proposer at round $r$, with $\text{support} + \text{noSupport} = \text{total}$.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{cert\_skip\_exclusive}]
Certificate and Skip are mutually exclusive: if $\text{total} \leq n$, then both cannot hold simultaneously.
\end{theorem}
\begin{proof}
Since $\text{support} + \text{noSupport} = \text{total} \leq n < 2(2f+1) = 4f+2$, both requiring $\geq 2f+1$ would need total $\geq 4f+2$.
\end{proof}

\begin{theorem}[\texttt{cert\_honest\_support}]
If a certificate exists ($\text{support} \geq 2f+1$) and at most $f$ supporters are Byzantine, then at least $f+1$ honest nodes support it.
\end{theorem}
\begin{proof}
By \texttt{omega}: $\text{honest} = \text{support} - \text{byzantine} \geq 2f+1-f = f+1$.
\end{proof}

\begin{theorem}[\texttt{classify\_total}]
Classification is exhaustive: every vote count yields exactly one of Commit, Skip, or Undecided.
\end{theorem}
