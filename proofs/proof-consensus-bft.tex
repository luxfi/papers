\author{Zach Kelling \and Woo Bin}

\begin{abstract}
We prove the fundamental Byzantine Fault Tolerance threshold properties that underpin the Lux consensus engine. The key result is that two quorums of size $> 2n/3$ must share at least one honest node, preventing conflicting certificates. All theorems are fully machine-checked in Lean~4 with Mathlib, using the \texttt{omega} tactic for arithmetic reasoning.
\end{abstract}

\section{Introduction}
The BFT threshold proof connects the standard assumption $f < n/3$ to the quorum intersection property used throughout the Lux consensus stack. This structural argument is referenced by Wave (threshold voting), Flare (DAG commit rule), and Quasar (hybrid quantum consensus).

\section{Definitions}
\begin{definition}[ConsensusParams]
Network of $n$ nodes with at most $f < n/3$ Byzantine. Quorum size $q > 2n/3$. Certificate threshold $2f+1$.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{quorum\_intersection}]
For $n$ nodes with $f < n/3$ Byzantine, two quorums $q_1, q_2 > 2n/3$ satisfy $q_1 + q_2 > n + f$.
\end{theorem}
\begin{proof}
By \texttt{omega} on the arithmetic constraints.
\end{proof}

\begin{theorem}[\texttt{honest\_majority\_sufficient}]
With $n = 3f+1$ nodes, the honest count $2f+1 > 2(3f+1)/3$.
\end{theorem}
\begin{proof}
Direct arithmetic via \texttt{omega}.
\end{proof}

\begin{theorem}[\texttt{unique\_alpha\_threshold}]
Two $\alpha$-quorums cannot both be satisfied by different values when $\alpha > 2k/3$, since total votes would exceed $k$.
\end{theorem}
\begin{proof}
By \texttt{omega}: $v_1 + v_2 \geq 2\alpha > 4k/3 > k$.
\end{proof}

\begin{theorem}[\texttt{certificate\_quorum\_overlap}]
Two certificate sets of size $\geq 2f+1$ from $n$ nodes with $3f < n$ overlap: $\text{cert}_1 + \text{cert}_2 > n$.
\end{theorem}
\begin{proof}
By \texttt{omega}.
\end{proof}

\begin{theorem}[\texttt{bft\_intersection}]
Key BFT property: $2(2f+1) > n + f$ when $3f < n$, ensuring any two subsets of size $2f+1$ share at least one honest node.
\end{theorem}
\begin{proof}
By \texttt{omega}.
\end{proof}
