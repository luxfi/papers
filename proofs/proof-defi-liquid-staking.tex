\title{Formal Verification: Liquid Staking and Yield Protocol}
\author{Zach Kelling, Lux Industries}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We formalize the liquid staking model where staked assets earn yield while remaining tradeable. The key properties are monotone yield accrual, no impermanent loss on single-sided staking, and compositional compounding.
\end{abstract}

\section{Introduction}
The liquid staking protocol enables ``money at rest'' to earn yield. Staked assets receive validator rewards while remaining tradeable via liquid derivative tokens (xLUX). Auto-compounding reinvests rewards into the staking position.

\section{Definitions}
\begin{definition}[Position]
Staking position: principal, accrued rewards, compounded rewards, liquid derivatives, duration.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{yield\_monotone}]
Accrued rewards only increase: $(\text{accrueBlock}~p).\text{accrued} \geq p.\text{accrued}$.
\end{theorem}

\begin{theorem}[\texttt{derivatives\_equal\_principal}]
Liquid derivatives are minted 1:1 with principal.
\end{theorem}

\begin{theorem}[\texttt{compound\_resets\_accrued}]
After compounding, accrued resets to 0.
\end{theorem}

\begin{theorem}[\texttt{compound\_increases\_base}]
Compounding increases the effective principal.
\end{theorem}

\begin{theorem}[\texttt{no\_impermanent\_loss}]
Single-sided staking has no impermanent loss: principal is unchanged by accrual.
\end{theorem}

\begin{theorem}[\texttt{value\_monotone}]
Total value (principal + compounded + accrued) is monotonically non-decreasing.
\end{theorem}
