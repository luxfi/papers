\title{Formal Verification: AMM Constant Product Invariant}
\author{Zach Kelling \and Woo Bin}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We formalize the constant-product AMM ($x \cdot y = k$) used in the Lux DEX. The key results are: the invariant is preserved (or increased by fees) on every swap, output is bounded by reserves, and zero input yields zero output.
\end{abstract}

\section{Introduction}
The AMM invariant proof ensures that the Lux DEX pool contracts never produce free tokens and that liquidity providers' positions are protected. The fee mechanism ensures that $k$ only grows over time, accruing value to LPs.

\section{Definitions}
\begin{definition}[Pool]
AMM pool state: reserves $x, y$, total LP tokens, fee in basis points.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{output\_bounded}]
Swap output is bounded by the reserve: $\text{dy} \leq \text{reserveY}$.
\end{theorem}

\begin{theorem}[\texttt{zero\_in\_zero\_out}]
Zero input yields zero output: $\text{swap}(p, 0).\text{dy} = 0$.
\end{theorem}

\begin{theorem}[\texttt{invariant\_preserved}]
The constant product $k = x \cdot y$ never decreases after a swap (fees increase it).
\end{theorem}
\begin{proof}
By \texttt{omega} on the fee-adjusted arithmetic.
\end{proof}

\begin{theorem}[\texttt{add\_liquidity\_grows}]
Adding liquidity increases reserves.
\end{theorem}

\begin{theorem}[\texttt{lp\_minted}]
LP tokens are minted on liquidity addition.
\end{theorem}
