\title{Formal Verification: Cross-Chain Compute Verification}
\author{Zach Kelling \and Woo Bin}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We prove that AI compute can be safely distributed across $N$ chains simultaneously with purely accretive rewards. The key properties are chain isolation (compute on chain $i$ doesn't affect chain $j$), reward monotonicity, conservation (rewards $\leq$ compute value), and commutativity (order of attestation processing doesn't matter).
\end{abstract}

\section{Introduction}
The cross-chain compute model enables distributed AI inference across the Lux multi-chain architecture. TEE attestations provide hardware-backed proof of compute, and the accretive reward model ensures that providers are always fairly compensated. The commutativity property ensures that the order of cross-chain attestation processing is irrelevant.

\section{Definitions}
\begin{definition}[ComputeAttestation]
Attestation from a single chain: chain ID, compute units, result hash, TEE validity, timestamp.
\end{definition}

\begin{definition}[ProviderState]
Provider state: total compute, total rewards, attestation list, active chains.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{chain\_isolation}]
Submitting compute on chain $i$ doesn't reduce existing compute from chain $j$: total compute only increases.
\end{theorem}
\begin{proof}
By case split on TEE validity.
\end{proof}

\begin{theorem}[\texttt{rewards\_accretive}]
Rewards only increase, never decrease.
\end{theorem}

\begin{theorem}[\texttt{invalid\_tee\_rejected}]
Without valid TEE attestation, state is unchanged.
\end{theorem}

\begin{theorem}[\texttt{reward\_bounded}]
Rewards never exceed compute value: $\text{reward} \leq \text{compute} \times \text{rate}$.
\end{theorem}
\begin{proof}
By \texttt{Nat.div\_le\_self}.
\end{proof}

\begin{theorem}[\texttt{multi\_chain\_commutative}]
Processing attestations from $N$ chains in any order yields the same total compute.
\end{theorem}
\begin{proof}
By commutativity of addition.
\end{proof}

\begin{theorem}[\texttt{chains\_monotone}]
Active chain count only grows.
\end{theorem}

\begin{theorem}[\texttt{zero\_compute\_zero\_reward}]
Zero compute yields zero reward.
\end{theorem}
