\begin{abstract}
We formalize Nebula, the DAG blockchain consensus mode that wraps Field. Nebula provides DAG blockchain semantics with topological ordering of committed vertices. Its correctness reduces to Field's consistent cut property.
\end{abstract}

\section{Introduction}
Nebula is the DAG consensus mode, composing as Nebula $\to$ Field $\to$ Wave $\to$ Prism $+$ Flare. It wraps Field and provides DAG blockchain semantics. All invariants delegate to Field's consistent cut property.

\section{Definitions}
\begin{definition}[Vertex]
A DAG vertex with id, parents, round, and author.
\end{definition}

\begin{definition}[NebulaState]
State: vertex set, committed set, and current round.
\end{definition}

\begin{definition}[IsTopologicalOrder]
Ordering property: if $u$ is a parent of $v$, then $u$ appears before $v$.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{committed\_monotone}]
Committed set only grows: $s.\text{committed} \subseteq s.\text{committed} \cup \text{newCommits}$.
\end{theorem}

\begin{theorem}[\texttt{vertices\_monotone}]
Vertex set only grows.
\end{theorem}

\begin{theorem}[\texttt{round\_monotone}]
Round number advances monotonically.
\end{theorem}
