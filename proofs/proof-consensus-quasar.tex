\title{Formal Verification: Quasar Hybrid Consensus}
\author{Zach Kelling \and Woo Bin}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We prove the soundness of the Quasar consensus engine, which combines BLS signature aggregation with Ringtail lattice-based threshold signatures for quantum-resistant finality. The key result is that a valid Quasar certificate requires both BLS AND Ringtail quorums, ensuring security even if one primitive is broken by a quantum computer.
\end{abstract}

\section{Introduction}
Quasar is the post-quantum finality engine for Lux. It collects blocks from all chains and produces quantum-final aggregate blocks with dual signatures. The hybrid construction ensures that an attacker must break both the BLS pairing assumption AND the LWE lattice assumption to forge certificates.

\section{Definitions}
\begin{definition}[QuasarCert]
A Quasar certificate containing round, value, BLS signature count, Ringtail signature count, and quorum threshold.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{hybrid\_requires\_both}]
A valid certificate requires both BLS and Ringtail signature counts to reach quorum.
\end{theorem}
\begin{proof}
Decompose the Boolean conjunction in \texttt{isValid}.
\end{proof}

\begin{theorem}[\texttt{quantum\_safe\_fallback}]
Even if BLS is broken by a quantum computer, Ringtail quorum remains genuine (lattice-based security).
\end{theorem}

\begin{theorem}[\texttt{two\_round\_finality}]
Finality is achieved in exactly 2 rounds: round $r$ proposes, round $r+1$ certifies.
\end{theorem}

\begin{theorem}[\texttt{cert\_intersection}]
Two valid certificates from $n$ validators with $3f < n$ satisfy $\text{blsSigs}_1 + \text{blsSigs}_2 > n$.
\end{theorem}
\begin{proof}
Decompose both certificates via \texttt{hybrid\_requires\_both}, then \texttt{omega}.
\end{proof}

\begin{theorem}[\texttt{unique\_per\_round}]
At most one value can be certified per round (from quorum intersection and honest majority).
\end{theorem}
