\title{Formal Verification: Teleport Bridge Protocol}
\author{Zach Kelling, Lux Industries}
\date{December 2025}

\begin{document}
\maketitle

\begin{abstract}
We formalize the lock-and-mint bridge protocol for the \$AI token with 7-of-11 ML-DSA-65 threshold signatures (per HIP-0101). The key properties are conservation (lock amount equals mint amount), threshold enforcement, and replay protection.
\end{abstract}

\section{Introduction}
The Teleport bridge enables cross-chain movement of the \$AI token. The 7-of-11 threshold ensures that no single validator can authorize a teleport, and quorum intersection prevents conflicting teleports for the same nonce.

\section{Definitions}
\begin{definition}[Chain]
A supported chain with id and name. Canonical root: Hanzo (chain 36963).
\end{definition}

\begin{definition}[TeleportOp]
A teleport operation: source, destination, amount, validator signatures, nonce.
\end{definition}

\begin{definition}[BridgeParams]
Bridge parameters: threshold $t=7$, validators $n=11$.
\end{definition}


\section{Theorems and Axioms}
\begin{theorem}[\texttt{teleport\_conserves}]
Conservation: locked amount on source equals minted amount on destination.
\end{theorem}
\begin{proof}
By \texttt{simp} on \texttt{executeTeleport}.
\end{proof}

\begin{theorem}[\texttt{insufficient\_sigs\_rejected}]
Below-threshold ($< 7$) teleports are rejected.
\end{theorem}

\begin{theorem}[\texttt{same\_chain\_rejected}]
Same-chain teleports are rejected.
\end{theorem}

\begin{theorem}[\texttt{zero\_amount\_rejected}]
Zero-amount teleports are rejected.
\end{theorem}

\begin{theorem}[\texttt{default\_bridge\_valid}]
Default bridge parameters: $t=7$, $n=11$.
\end{theorem}

\begin{theorem}[\texttt{replay\_protection}]
Different nonces yield different operations.
\end{theorem}

\begin{theorem}[\texttt{finality\_from\_quorum}]
Two valid teleports with the same nonce have overlapping validator sets: $\text{sigs}_1 + \text{sigs}_2 > n$.
\end{theorem}
\begin{proof}
By \texttt{omega} with the threshold constraint.
\end{proof}
