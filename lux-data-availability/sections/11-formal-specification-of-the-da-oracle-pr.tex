\section{Formal Specification of the DA Oracle Protocol}

\label{app:oracle}

\begin{algorithm}[H]
\caption{DA Oracle Attestation Protocol}
\label{alg:da-oracle}
\begin{algorithmic}[1]
\Require Full node $v_i$ with signing key $sk_i$, block header $H$
\Ensure DA attestation or rejection
\State Download all cells of $\hat{M}$ from block producer and peers.
\State Verify all $4k^2$ KZG cell proofs against $H.\text{DACommitment}$.
\If{any cell proof fails}
    \State Broadcast fraud proof $(i, j, \hat{M}[i][j], \pi_{\text{invalid}})$.
    \State \Return \textbf{Reject}
\EndIf
\State Verify 2D RS consistency: for each row $i$, check that $\hat{R}_i$ has degree $< k$.
\If{any row degree check fails}
    \State Broadcast bad-encoding fraud proof for row $i$.
    \State \Return \textbf{Reject}
\EndIf
\State Sign attestation: $\sigma_i \gets \text{BLS.Sign}(sk_i, H.\text{DataRoot})$.
\State Broadcast $\sigma_i$ on the DA gossip topic.
\State \Return \textbf{Accept}
\end{algorithmic}
\end{algorithm}
