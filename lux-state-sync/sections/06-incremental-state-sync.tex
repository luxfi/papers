\section{Incremental State Sync}
\label{sec:incremental}

\subsection{Motivation}

Nodes that are temporarily offline (maintenance, network issues) need not re-download the entire state. Incremental sync transfers only the state delta between the node's last known state root $R_{\text{old}}$ and the current target $R_{\text{new}}$.

\subsection{State Diff Computation}

\begin{definition}[State Diff]
The state diff $\Delta(R_{\text{old}}, R_{\text{new}})$ consists of:
\begin{itemize}
    \item Inserted keys: $\{k : k \in \text{Trie}(R_{\text{new}}) \setminus \text{Trie}(R_{\text{old}})\}$
    \item Deleted keys: $\{k : k \in \text{Trie}(R_{\text{old}}) \setminus \text{Trie}(R_{\text{new}})\}$
    \item Modified keys: $\{k : V_{R_{\text{old}}}(k) \neq V_{R_{\text{new}}}(k)\}$
\end{itemize}
\end{definition}

\begin{algorithm}[H]
\caption{Incremental Sync Protocol}
\label{alg:incremental}
\begin{algorithmic}[1]
\Require Local state root $R_{\text{old}}$, target root $R_{\text{new}}$
\State Request $\Delta(R_{\text{old}}, R_{\text{new}})$ from peer
\State Peer computes diff by traversing both tries simultaneously
\State Peer sends $\Delta$ with range proofs anchored at $R_{\text{new}}$
\State Apply insertions, deletions, and modifications to local trie
\State Recompute local root $R' \gets H(\text{updated trie})$
\If{$R' = R_{\text{new}}$}
    \State Sync complete
\Else
    \State Fall back to full state sync
\EndIf
\end{algorithmic}
\end{algorithm}

\subsection{Diff Size Analysis}

\begin{proposition}[Incremental Diff Size]
For a trie with $n$ leaves and $m$ modifications over $\tau$ blocks, the diff size is:
\begin{equation}
    |\Delta| = O(m \cdot (L + d \cdot 32))
\end{equation}
where $L$ is the average leaf size and $d$ is the trie depth. The diff is independent of total state size $n$.
\end{proposition}

\subsection{Diff Verification}

The incremental diff must be verified for correctness and completeness:

\begin{theorem}[Diff Verification]
An incremental diff $\Delta$ with proofs anchored at $R_{\text{new}}$ is verifiable if:
\begin{enumerate}
    \item Each inserted/modified leaf is accompanied by a Merkle inclusion proof in $R_{\text{new}}$.
    \item Each deleted leaf is accompanied by a Merkle exclusion proof in $R_{\text{new}}$.
    \item Applying $\Delta$ to $\text{Trie}(R_{\text{old}})$ produces root $R_{\text{new}}$.
\end{enumerate}
\end{theorem}
