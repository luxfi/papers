\subsection{Appchain Creation and Management}



\begin{algorithmic}[1]
\State \textbf{function} \textsc{CreateAppchain}(owners, threshold, controlKeys)
    \State $blsSig \gets \text{SignBLS}(owners, controlKeys)$
    \State $rtSig \gets \text{SignRingtail}(owners, controlKeys)$
    \State $tripleProof \gets \text{TripleProof}(blsSig, rtSig)$
    \If{$\text{Verify}(tripleProof)$}
        \State $appchain \gets \text{AllocateAppchain}(owners, threshold)$
        \State \Return $appchain$
    \EndIf
    \State \Return $\texttt{INVALID}$
\State \textbf{end function}
\end{algorithmic}
